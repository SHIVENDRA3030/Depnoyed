    74→function statusBannerStyle(status: string): { bg: string; icon: React.ReactNode; text: string } {
    75→  switch (status) {
    76→    case "running":
    77→      return {
    78→        bg: "bg-emerald-50 border-emerald-200 dark:bg-emerald-950/40 dark:border-emerald-800",
    79→        icon: <CheckCircle2 className="size-5 text-emerald-600 dark:text-emerald-400" />,
    80→        text: "This deployment is running and accessible.",
    81→      };
    82→    case "pending":
    83→    case "creating":
    84→      return {
    85→        bg: "bg-amber-50 border-amber-200 dark:bg-amber-950/40 dark:border-amber-800",
    86→        icon: <Loader2 className="size-5 animate-spin text-amber-600 dark:text-amber-400" />,
    87→        text: "Deployment is being provisioned…",
    88→      };
    89→    case "failed":
    90→    case "dead":
    91→      return {
    92→        bg: "bg-red-50 border-red-200 dark:bg-red-950/40 dark:border-red-800",
    93→        icon: <AlertCircle className="size-5 text-red-600 dark:text-red-400" />,
    94→        text: "Deployment failed. Try restarting or delete and re-deploy.",
    95→      };
    96→    case "stopped":
    97→    case "exited":
    98→      return {
    99→        bg: "bg-zinc-50 border-zinc-200 dark:bg-zinc-900/40 dark:border-zinc-700",
   100→        icon: <Square className="size-5 text-zinc-500 dark:text-zinc-400" />,
   101→        text: "Deployment is stopped. Start it to make it accessible.",
   102→      };
   103→    default:
   104→      return {
   105→        bg: "bg-muted border-border",
   106→        icon: <Circle className="size-5 text-muted-foreground" />,
   107→        text: "Unknown status.",
   108→      };
   109→  }
   110→}

   139→      /* ignore */
   140→    }

   148→      const keys = res.result.keys ?? [];

   150→      // Fetch values for each key (limit to first 20 keys to avoid overload).

   155→            const r = await api<{ result: { ok: boolean; value?: string | null } }>(
   156→              `/api/deployments/${id}/volume?op=get&key=${encodeURIComponent(k)}`
   157→            );
   158→            return [k, r.result.value ?? ""] as const;
   159→          } catch {
   160→            return [k, ""] as const;
   161→          }
   162→        })
   163→      );
   164→      setVolumeValues(Object.fromEntries(entries));
   165→    } catch {
   166→      setVolumeKeys([]);
   167→    } finally {
   168→      setVolumeLoading(false);
   169→    }

   196→      toast.error(e instanceof ApiError ? e.message : `Failed to ${action}`);
   197→    } finally {
   198→      setBusy(false);
   199→    }

   256→

   345→            </div>

   370→        <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">

   372→            <AppLogo logo={dep.app?.logo ?? null} simulator={dep.app?.simulator ?? "static"} name={dep.app?.name ?? "App"} size="lg" />

   430→                  >

   768→function computeSteps(status: string): { label: string; state: "done" | "active" | "pending" }[] {
   769→  const labels = [
   770→    "Creating deployment",
   771→    "Creating volume",
   772→    "Creating container",
   773→    "Starting application",
   774→    "Deployment ready",
   775→  ];
   776→  const order: Record<string, number> = {
   777→    pending: 0,
   778→    creating: 1,
   779→    created: 2,
   780→    running: 5,
   781→    stopped: 5,
   782→    exited: 5,
   783→    failed: 5,
   784→    dead: 5,
   785→  };
   786→  const progress = order[status] ?? 0;
   787→  return labels.map((label, i) => {
   788→    if (progress >= 5) return { label, state: "done" as const };
   789→    if (i < progress) return { label, state: "done" as const };
   790→    if (i === progress) return { label, state: "active" as const };
   791→    return { label, state: "pending" as const };
   792→  });
   793→}

   795→function timeAgo(iso: string): string {
   796→  const diff = Date.now() - new Date(iso).getTime();
   797→  const mins = Math.floor(diff / 60000);
   798→  if (mins < 1) return "just now";
   799→  if (mins < 60) return `${mins}m ago`;
   800→  const hrs = Math.floor(mins / 60);
   801→  if (hrs < 24) return `${hrs}h ago`;
   802→  const days = Math.floor(hrs / 24);
   803→  return `${days}d ago`;
   804→}

   806→function uptimeSince(iso: string): string {
   807→  const diff = Date.now() - new Date(iso).getTime();
   808→  const mins = Math.floor(diff / 60000);
   809→  if (mins < 1) return "<1m";
   810→  if (mins < 60) return `${mins}m`;
   811→  const hrs = Math.floor(mins / 60);
   812→  const remainMins = mins % 60;
   813→  if (hrs < 24) return remainMins > 0 ? `${hrs}h ${remainMins}m` : `${hrs}h`;
   814→  const days = Math.floor(hrs / 24);
   815→  return `${days}d`;
   816→}

   822→function DeploymentPerformanceSparklines({
   823→  containerId,
   824→  status,
   825→  metrics,
   826→}: {
   827→  containerId: string;
   828→  status: string;
   829→  metrics: { cpuUsagePercent: number; memoryUsagePercent: number; responseLatencyMs: number; healthScore: number };
   830→}) {
   831→  const running = status === "running";
   832→  const color = sparklineColor(status);
   833→
   834→  // Base metrics from getContainerMetrics
   835→  const baseCpu = metrics.cpuUsagePercent;
   836→  const baseMemory = metrics.memoryUsagePercent;
   837→  const baseNetwork = metrics.responseLatencyMs / 150 * 40 + 10; // ~10-50
   838→  const baseDisk = metrics.healthScore * 0.3 + 5; // ~5-35
   839→
   840→  // Tick counter drives series evolution (only ticks for running deployments)
   841→  const [tick, setTick] = useState(0);
   842→
   843→  // Auto-refresh every 5 seconds for running deployments
   844→  useEffect(() => {
   845→    if (!running) return;
   846→    const t = setInterval(() => setTick((n) => n + 1), 5000);
   847→    return () => clearInterval(t);
   848→  }, [running]);
   849→
   850→  // Compute full series from containerId + tick (fully derived)
   851→  const series = useMemo(() => {
   852→    const cpu = generateTimeSeries(baseCpu, DEPLOY_SPARKLINE_POINTS, 4, `dep-${containerId}:cpu`);
   853→    const memory = generateTimeSeries(baseMemory, DEPLOY_SPARKLINE_POINTS, 3, `dep-${containerId}:mem`);
   854→    const network = generateTimeSeries(baseNetwork, DEPLOY_SPARKLINE_POINTS, 5, `dep-${containerId}:net`);
   855→    const disk = generateTimeSeries(baseDisk, DEPLOY_SPARKLINE_POINTS, 3, `dep-${containerId}:disk`);
   856→    // Apply tick-based evolution
   857→    let c = cpu, m = memory, n = network, d = disk;
   858→    for (let i = 0; i < tick; i++) {
   859→      c = tickTimeSeries(c, baseCpu, 4, i + 1);
   860→      m = tickTimeSeries(m, baseMemory, 3, i + 1);
   861→      n = tickTimeSeries(n, baseNetwork, 5, i + 1);
   862→      d = tickTimeSeries(d, baseDisk, 3, i + 1);
   863→    }
   864→    return { cpu: c, memory: m, network: n, disk: d };
   865→  }, [containerId, baseCpu, baseMemory, baseNetwork, baseDisk, tick]);
   866→
   867→  const charts: {
   868→    key: string;
   869→    label: string;
   870→    icon: React.ReactNode;
   871→    data: number[];
   872→    unit: string;
   873→    decimals?: number;
   874→  }[] = [
   875→    {
   876→      key: "cpu",
   877→      label: "CPU Usage",
   878→      icon: <Cpu className="size-4" />,
   879→      data: series.cpu,
   880→      unit: "%",
   881→    },
   882→    {
   883→      key: "memory",

   885→      icon: <MemoryStick className="size-4" />,

   888→    },
   889→    {
   890→      key: "network",
   891→      label: "Network I/O",
   892→      icon: <Wifi className="size-4" />,
   893→      data: series.network,
   894→      unit: " MB/s",
   895→      decimals: 1,
   896→    },
   897→    {
   898→      key: "disk",
   899→      label: "Disk I/O",
   900→      icon: <HardDriveUpload className="size-4" />,
   901→      data: series.disk,
   902→      unit: " MB/s",
   903→      decimals: 1,
   904→    },
   905→  ];
   906→
   907→  return (
   908→    <div className="mb-6 rounded-2xl border border-border bg-card p-5 shadow-sm">
   909→      <h2 className="flex items-center gap-2 text-sm font-semibold">
   910→        <Activity className="size-4 text-brand" /> Performance
   911→      </h2>
   912→      <div className="mt-4 grid grid-cols-2 gap-4 sm:grid-cols-4">
   913→        {charts.map((c) => {
   914→          const current = c.data[c.data.length - 1] ?? 0;
   915→          const prev = c.data[c.data.length - 2] ?? current;
   916→          const trendUp = current >= prev;
   917→          const min = Math.min(...c.data);
   918→          const max = Math.max(...c.data);

   920→          const displayMin = c.decimals ? min.toFixed(c.decimals) : Math.round(min);
