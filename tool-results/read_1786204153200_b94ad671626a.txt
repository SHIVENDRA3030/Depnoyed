    74→    74→function statusBannerStyle(status: string): { bg: string; icon: React.ReactNode; text: string } {
    75→    75→  switch (status) {
    76→    76→    case "running":
    77→    77→      return {
    78→    78→        bg: "bg-emerald-50 border-emerald-200 dark:bg-emerald-950/40 dark:border-emerald-800",
    79→    79→        icon: <CheckCircle2 className="size-5 text-emerald-600 dark:text-emerald-400" />,
    80→    80→        text: "This deployment is running and accessible.",
    81→    81→      };
    82→    82→    case "pending":
    83→    83→    case "creating":
    84→    84→      return {
    85→    85→        bg: "bg-amber-50 border-amber-200 dark:bg-amber-950/40 dark:border-amber-800",
    86→    86→        icon: <Loader2 className="size-5 animate-spin text-amber-600 dark:text-amber-400" />,
    87→    87→        text: "Deployment is being provisioned…",
    88→    88→      };
    89→    89→    case "failed":
    90→    90→    case "dead":
    91→    91→      return {
    92→    92→        bg: "bg-red-50 border-red-200 dark:bg-red-950/40 dark:border-red-800",
    93→    93→        icon: <AlertCircle className="size-5 text-red-600 dark:text-red-400" />,
    94→    94→        text: "Deployment failed. Try restarting or delete and re-deploy.",
    95→    95→      };
    96→    96→    case "stopped":

   101→   101→        text: "Deployment is stopped. Start it to make it accessible.",

   105→   105→        bg: "bg-muted border-border",
   106→   106→        icon: <Circle className="size-5 text-muted-foreground" />,

   768→   768→function computeSteps(status: string): { label: string; state: "done" | "active" | "pending" }[] {
   769→   769→  const labels = [
   770→   770→    "Creating deployment",
   771→   771→    "Creating volume",
   772→   772→    "Creating container",
   773→   773→    "Starting application",
   774→   774→    "Deployment ready",
   775→   775→  ];
   776→   776→  const order: Record<string, number> = {
   777→   777→    pending: 0,
   778→   778→    creating: 1,
   779→   779→    created: 2,
   780→   780→    running: 5,
   781→   781→    stopped: 5,
   782→   782→    exited: 5,
   783→   783→    failed: 5,
   784→   784→    dead: 5,
   785→   785→  };
   786→   786→  const progress = order[status] ?? 0;
   787→   787→  return labels.map((label, i) => {
   788→   788→    if (progress >= 5) return { label, state: "done" as const };
   789→   789→    if (i < progress) return { label, state: "done" as const };
   790→   790→    if (i === progress) return { label, state: "active" as const };
   791→   791→    return { label, state: "pending" as const };
   792→   792→  });
   793→   793→}

   806→   806→function uptimeSince(iso: string): string {
   807→   807→  const diff = Date.now() - new Date(iso).getTime();

   809→   809→  if (mins < 1) return "<1m";
   810→   810→  if (mins < 60) return `${mins}m`;
   811→   811→  const hrs = Math.floor(mins / 60);

   813→   813→  if (hrs < 24) return remainMins > 0 ? `${hrs}h ${remainMins}m` : `${hrs}h`;
   814→   814→  const days = Math.floor(hrs / 24);
   815→   815→  return `${days}d`;

   822→   822→function DeploymentPerformanceSparklines({

   827→   827→  containerId: string;

   829→   829→  metrics: { cpuUsagePercent: number; memoryUsagePercent: number; responseLatencyMs: number; healthScore: number };
   830→   830→}) {

   832→   832→  const color = sparklineColor(status);
   833→   833→
   834→   834→  // Base metrics from getContainerMetrics
   835→   835→  const baseCpu = metrics.cpuUsagePercent;
   836→   836→  const baseMemory = metrics.memoryUsagePercent;
   837→   837→  const baseNetwork = metrics.responseLatencyMs / 150 * 40 + 10; // ~10-50
   838→   838→  const baseDisk = metrics.healthScore * 0.3 + 5; // ~5-35
   839→   839→
   840→   840→  // Tick counter drives series evolution (only ticks for running deployments)
   841→   841→  const [tick, setTick] = useState(0);
   842→   842→
   843→   843→  // Auto-refresh every 5 seconds for running deployments
   844→   844→  useEffect(() => {
   845→   845→    if (!running) return;
   846→   846→    const t = setInterval(() => setTick((n) => n + 1), 5000);
   847→   847→    return () => clearInterval(t);
   848→   848→  }, [running]);
   849→   849→
   850→   850→  // Compute full series from containerId + tick (fully derived)
   851→   851→  const series = useMemo(() => {
   852→   852→    const cpu = generateTimeSeries(baseCpu, DEPLOY_SPARKLINE_POINTS, 4, `dep-${containerId}:cpu`);
   853→   853→    const memory = generateTimeSeries(baseMemory, DEPLOY_SPARKLINE_POINTS, 3, `dep-${containerId}:mem`);
   854→   854→    const network = generateTimeSeries(baseNetwork, DEPLOY_SPARKLINE_POINTS, 5, `dep-${containerId}:net`);
   855→   855→    const disk = generateTimeSeries(baseDisk, DEPLOY_SPARKLINE_POINTS, 3, `dep-${containerId}:disk`);
   856→   856→    // Apply tick-based evolution
   857→   857→    let c = cpu, m = memory, n = network, d = disk;
   858→   858→    for (let i = 0; i < tick; i++) {
   859→   859→      c = tickTimeSeries(c, baseCpu, 4, i + 1);
   860→   860→      m = tickTimeSeries(m, baseMemory, 3, i + 1);
   861→   861→      n = tickTimeSeries(n, baseNetwork, 5, i + 1);
   862→   862→      d = tickTimeSeries(d, baseDisk, 3, i + 1);
   863→   863→    }
   864→   864→    return { cpu: c, memory: m, network: n, disk: d };
   865→   865→  }, [containerId, baseCpu, baseMemory, baseNetwork, baseDisk, tick]);
   866→   866→
   867→   867→  const charts: {
   868→   868→    key: string;
