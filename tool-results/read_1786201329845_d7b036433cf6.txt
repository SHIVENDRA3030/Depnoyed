   850→   850→function timeAgo(iso: string): string {
   851→   851→  const diff = Date.now() - new Date(iso).getTime();
   852→   852→  const mins = Math.floor(diff / 60000);
   853→   853→  if (mins < 1) return "just now";
   854→   854→  if (mins < 60) return `${mins}m ago`;
   855→   855→  const hrs = Math.floor(mins / 60);
   856→   856→  if (hrs < 24) return `${hrs}h ago`;
   857→   857→  const days = Math.floor(hrs / 24);
   858→   858→  return `${days}d ago`;
   859→   859→}

   861→   861→function uptimeSince(iso: string): string {
   862→   862→  const diff = Date.now() - new Date(iso).getTime();
   863→   863→  const mins = Math.floor(diff / 60000);
   864→   864→  if (mins < 1) return "<1m";
   865→   865→  if (mins < 60) return `${mins}m`;
   866→   866→  const hrs = Math.floor(mins / 60);
   867→   867→  const remainMins = mins % 60;
   868→   868→  if (hrs < 24) return remainMins > 0 ? `${hrs}h ${remainMins}m` : `${hrs}h`;
   869→   869→  const days = Math.floor(hrs / 24);
   870→   870→  return `${days}d`;
   871→   871→}

   873→   873→function formatDataSize(bytes: number): string {
   874→   874→  if (bytes < 1024) return `${bytes} B`;
   875→   875→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
   876→   876→  return `${(bytes / (1024 * 1024)).toFixed(1)} MB`;
   877→   877→}

   879→   879→function estimateCost(deployments: DeploymentItem[]) {
   880→   880→  const running = deployments.filter((d) => d.status === "running").length;
   881→   881→  const cpuCost = 0.5 * running * 5;  // $5/core/month
   882→   882→  const memCost = 0.5 * running * 3;   // $3/512MB/month
   883→   883→  const totalStorage = deployments.reduce((a, d) => a + (d.volumeDataSize ?? 0), 0);
   884→   884→  const storageGB = Math.max(1, totalStorage / 1024 / 1024);
   885→   885→  const storageCost = storageGB * 0.10; // $0.10/GB/month
   886→   886→  const monthly = cpuCost + memCost + storageCost;
   887→   887→  const daily = monthly / 30;
   888→   888→  return { monthly, daily };
   889→   889→}

   897→   897→function ResourceUsageSparklines({ deployments }: { deployments: DeploymentItem[] }) {
   898→   898→  const runningDeployments = deployments.filter((d) => d.status === "running");
   899→   899→  const hasRunning = runningDeployments.length > 0;
   900→   900→
   901→   901→  // Compute aggregate base metrics across all running deployments
   902→   902→  const aggMetrics = useMemo(() => {
   903→   903→    if (runningDeployments.length === 0) {
   904→   904→      return { cpu: 0, memory: 0, network: 0, disk: 0 };
   905→   905→    }
   906→   906→    let cpu = 0;
   907→   907→    let memory = 0;
   908→   908→    let network = 0;
   909→   909→    let disk = 0;
   910→   910→    for (const d of runningDeployments) {
   911→   911→      const containerId = d.containerId ?? d.id;
   912→   912→      const m = getContainerMetrics(containerId);
   913→   913→      cpu += m.cpuUsagePercent;
   914→   914→      memory += m.memoryUsagePercent;
   915→   915→      // Derive network/disk from other metrics for realism
   916→   916→      network += m.responseLatencyMs / 150 * 40 + 10; // ~10-50 range
   917→   917→      disk += m.healthScore * 0.3 + 5; // ~5-35 range
   918→   918→    }
   919→   919→    const n = runningDeployments.length;
   920→   920→    return {
   921→   921→      cpu: cpu / n,
   922→   922→      memory: memory / n,
   923→   923→      network: network / n,
   924→   924→      disk: disk / n,
   925→   925→    };
   926→   926→  }, [runningDeployments]);
   927→   927→
   928→   928→  // Stable seed for initial series generation
   929→   929→  const seed = useMemo(
   930→   930→    () => `dash-${runningDeployments.map((d) => d.id).join(",")}`,
   931→   931→    [runningDeployments],
   932→   932→  );
   933→   933→
   934→   934→  // Tick counter drives series evolution
   935→   935→  const [tick, setTick] = useState(0);
   936→   936→
   937→   937→  // Auto-refresh every 10 seconds

   939→   939→    const t = setInterval(() => setTick((n) => n + 1), 10000);
   940→   940→    return () => clearInterval(t);
   941→   941→  }, []);
   942→   942→
   943→   943→  // Compute full series from seed + tick (fully derived, no intermediate state)
   944→   944→  const series = useMemo(() => {
   945→   945→    const cpu = generateTimeSeries(aggMetrics.cpu, SPARKLINE_POINTS, 5, `${seed}:cpu`);
   946→   946→    const memory = generateTimeSeries(aggMetrics.memory, SPARKLINE_POINTS, 4, `${seed}:mem`);
   947→   947→    const network = generateTimeSeries(aggMetrics.network, SPARKLINE_POINTS, 6, `${seed}:net`);
   948→   948→    const disk = generateTimeSeries(aggMetrics.disk, SPARKLINE_POINTS, 4, `${seed}:disk`);
   949→   949→    // Apply tick-based evolution

   953→   953→      m = tickTimeSeries(m, aggMetrics.memory, 4, i + 1);
   954→   954→      n = tickTimeSeries(n, aggMetrics.network, 6, i + 1);
   955→   955→      d = tickTimeSeries(d, aggMetrics.disk, 4, i + 1);
   956→   956→    }
