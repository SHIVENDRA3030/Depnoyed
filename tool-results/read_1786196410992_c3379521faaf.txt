    73→function statusBannerStyle(status: string): { bg: string; icon: React.ReactNode; text: string } {
    74→  switch (status) {
    75→    case "running":
    76→      return {
    77→        bg: "bg-emerald-50 border-emerald-200 dark:bg-emerald-950/40 dark:border-emerald-800",
    78→        icon: <CheckCircle2 className="size-5 text-emerald-600 dark:text-emerald-400" />,
    79→        text: "This deployment is running and accessible.",
    80→      };
    81→    case "pending":
    82→    case "creating":
    83→      return {
    84→        bg: "bg-amber-50 border-amber-200 dark:bg-amber-950/40 dark:border-amber-800",
    85→        icon: <Loader2 className="size-5 animate-spin text-amber-600 dark:text-amber-400" />,
    86→        text: "Deployment is being provisioned…",
    87→      };
    88→    case "failed":
    89→    case "dead":
    90→      return {
    91→        bg: "bg-red-50 border-red-200 dark:bg-red-950/40 dark:border-red-800",
    92→        icon: <AlertCircle className="size-5 text-red-600 dark:text-red-400" />,
    93→        text: "Deployment failed. Try restarting or delete and re-deploy.",
    94→      };
    95→    case "stopped":
    96→    case "exited":
    97→      return {
    98→        bg: "bg-zinc-50 border-zinc-200 dark:bg-zinc-900/40 dark:border-zinc-700",
    99→        icon: <Square className="size-5 text-zinc-500 dark:text-zinc-400" />,
   100→        text: "Deployment is stopped. Start it to make it accessible.",
   101→      };
   102→    default:
   103→      return {
   104→        bg: "bg-muted border-border",
   105→        icon: <Circle className="size-5 text-muted-foreground" />,
   106→        text: "Unknown status.",
   107→      };
   108→  }
   109→}

   130→    }

   138→      /* ignore */
   139→    }

   147→      const keys = res.result.keys ?? [];

   149→      // Fetch values for each key (limit to first 20 keys to avoid overload).
   150→      const limited = keys.slice(0, 20);

   152→        limited.map(async (k) => {

   154→            const r = await api<{ result: { ok: boolean; value?: string | null } }>(
   155→              `/api/deployments/${id}/volume?op=get&key=${encodeURIComponent(k)}`
   156→            );
   157→            return [k, r.result.value ?? ""] as const;
   158→          } catch {
   159→            return [k, ""] as const;
   160→          }
   161→        })
   162→      );
   163→      setVolumeValues(Object.fromEntries(entries));
   164→    } catch {
   165→      setVolumeKeys([]);
   166→    } finally {
   167→      setVolumeLoading(false);
   168→    }

   194→    } catch (e) {
   195→      toast.error(e instanceof ApiError ? e.message : `Failed to ${action}`);
   196→    } finally {
   197→      setBusy(false);
   198→    }

   255→

   735→function computeSteps(status: string): { label: string; state: "done" | "active" | "pending" }[] {
   736→  const labels = [
   737→    "Creating deployment",
   738→    "Creating volume",
   739→    "Creating container",
   740→    "Starting application",
   741→    "Deployment ready",
   742→  ];
   743→  const order: Record<string, number> = {
   744→    pending: 0,
   745→    creating: 1,
   746→    created: 2,
   747→    running: 5,
   748→    stopped: 5,
   749→    exited: 5,
   750→    failed: 5,
   751→    dead: 5,
   752→  };
   753→  const progress = order[status] ?? 0;
   754→  return labels.map((label, i) => {
   755→    if (progress >= 5) return { label, state: "done" as const };
   756→    if (i < progress) return { label, state: "done" as const };
   757→    if (i === progress) return { label, state: "active" as const };
   758→    return { label, state: "pending" as const };
   759→  });
   760→}

   762→function timeAgo(iso: string): string {
   763→  const diff = Date.now() - new Date(iso).getTime();
   764→  const mins = Math.floor(diff / 60000);
   765→  if (mins < 1) return "just now";
   766→  if (mins < 60) return `${mins}m ago`;
   767→  const hrs = Math.floor(mins / 60);
   768→  if (hrs < 24) return `${hrs}h ago`;
   769→  const days = Math.floor(hrs / 24);
   770→  return `${days}d ago`;
   771→}

   773→function uptimeSince(iso: string): string {
   774→  const diff = Date.now() - new Date(iso).getTime();
   775→  const mins = Math.floor(diff / 60000);
   776→  if (mins < 1) return "<1m";
   777→  if (mins < 60) return `${mins}m`;
   778→  const hrs = Math.floor(mins / 60);
   779→  const remainMins = mins % 60;
   780→  if (hrs < 24) return remainMins > 0 ? `${hrs}h ${remainMins}m` : `${hrs}h`;
   781→  const days = Math.floor(hrs / 24);
   782→  return `${days}d`;
   783→}

   789→function DeploymentPerformanceSparklines({
   790→  containerId,
   791→  status,
   792→  metrics,
   793→}: {
   794→  containerId: string;
   795→  status: string;
   796→  metrics: { cpuUsagePercent: number; memoryUsagePercent: number; responseLatencyMs: number; healthScore: number };
   797→}) {
   798→  const running = status === "running";
   799→  const color = sparklineColor(status);
   800→
   801→  // Base metrics from getContainerMetrics
   802→  const baseCpu = metrics.cpuUsagePercent;
   803→  const baseMemory = metrics.memoryUsagePercent;
   804→  const baseNetwork = metrics.responseLatencyMs / 150 * 40 + 10; // ~10-50
   805→  const baseDisk = metrics.healthScore * 0.3 + 5; // ~5-35
   806→
   807→  // Tick counter drives series evolution (only ticks for running deployments)
   808→  const [tick, setTick] = useState(0);
   809→
   810→  // Auto-refresh every 5 seconds for running deployments
   811→  useEffect(() => {
   812→    if (!running) return;
   813→    const t = setInterval(() => setTick((n) => n + 1), 5000);
   814→    return () => clearInterval(t);
   815→  }, [running]);
   816→
   817→  // Compute full series from containerId + tick (fully derived)
   818→  const series = useMemo(() => {
   819→    const cpu = generateTimeSeries(baseCpu, DEPLOY_SPARKLINE_POINTS, 4, `dep-${containerId}:cpu`);
   820→    const memory = generateTimeSeries(baseMemory, DEPLOY_SPARKLINE_POINTS, 3, `dep-${containerId}:mem`);
   821→    const network = generateTimeSeries(baseNetwork, DEPLOY_SPARKLINE_POINTS, 5, `dep-${containerId}:net`);
   822→    const disk = generateTimeSeries(baseDisk, DEPLOY_SPARKLINE_POINTS, 3, `dep-${containerId}:disk`);
   823→    // Apply tick-based evolution
   824→    let c = cpu, m = memory, n = network, d = disk;
   825→    for (let i = 0; i < tick; i++) {
   826→      c = tickTimeSeries(c, baseCpu, 4, i + 1);
   827→      m = tickTimeSeries(m, baseMemory, 3, i + 1);
   828→      n = tickTimeSeries(n, baseNetwork, 5, i + 1);
   829→      d = tickTimeSeries(d, baseDisk, 3, i + 1);
   830→    }
   831→    return { cpu: c, memory: m, network: n, disk: d };
   832→  }, [containerId, baseCpu, baseMemory, baseNetwork, baseDisk, tick]);
   833→
   834→  const charts: {
   835→    key: string;
   836→    label: string;
   837→    icon: React.ReactNode;
   838→    data: number[];
   839→    unit: string;
   840→    decimals?: number;
   841→  }[] = [
   842→    {
   843→      key: "cpu",
   844→      label: "CPU Usage",
   845→      icon: <Cpu className="size-4" />,
   846→      data: series.cpu,
   847→      unit: "%",
   848→    },
   849→    {
   850→      key: "memory",
   851→      label: "Memory Usage",
   852→      icon: <MemoryStick className="size-4" />,
   853→      data: series.memory,
   854→      unit: "%",
   855→    },
   856→    {
   857→      key: "network",
   858→      label: "Network I/O",
   859→      icon: <Wifi className="size-4" />,
   860→      data: series.network,
   861→      unit: " MB/s",
   862→      decimals: 1,
   863→    },
   864→    {
   865→      key: "disk",
   866→      label: "Disk I/O",
   867→      icon: <HardDriveUpload className="size-4" />,
   868→      data: series.disk,
   869→      unit: " MB/s",
   870→      decimals: 1,
   871→    },
   872→  ];
   873→
   874→  return (
   875→    <div className="mb-6 rounded-2xl border border-border bg-card p-5 shadow-sm">
   876→      <h2 className="flex items-center gap-2 text-sm font-semibold">
   877→        <Activity className="size-4 text-brand" /> Performance
   878→      </h2>
   879→      <div className="mt-4 grid grid-cols-2 gap-4 sm:grid-cols-4">
   880→        {charts.map((c) => {
   881→          const current = c.data[c.data.length - 1] ?? 0;
   882→          const prev = c.data[c.data.length - 2] ?? current;
   883→          const trendUp = current >= prev;
   884→          const min = Math.min(...c.data);
   885→          const max = Math.max(...c.data);
   886→          const displayVal = c.decimals ? current.toFixed(c.decimals) : Math.round(current);
   887→          const displayMin = c.decimals ? min.toFixed(c.decimals) : Math.round(min);
   888→          const displayMax = c.decimals ? max.toFixed(c.decimals) : Math.round(max);
   889→          return (
   890→            <div
   891→              key={c.key}
   892→              className="flex flex-col rounded-xl border border-border bg-muted/30 p-3"
   893→            >
   894→              <div className="flex items-center justify-between">
   895→                <span className="flex items-center gap-1.5 text-[11px] font-medium uppercase tracking-wider text-muted-foreground">
   896→                  {c.icon} {c.label}
   897→                </span>

   899→                  {trendUp ? <ArrowUpRight className="size-3" /> : <ArrowDownRight className="size-3" />}
