    76→function statusBannerStyle(status: string): { bg: string; icon: React.ReactNode; text: string } {
    77→  switch (status) {
    78→    case "running":
    79→      return {
    80→        bg: "bg-emerald-50 border-emerald-200 dark:bg-emerald-950/40 dark:border-emerald-800",
    81→        icon: <CheckCircle2 className="size-5 text-emerald-600 dark:text-emerald-400" />,
    82→        text: "This deployment is running and accessible.",
    83→      };
    84→    case "pending":
    85→    case "creating":
    86→      return {
    87→        bg: "bg-amber-50 border-amber-200 dark:bg-amber-950/40 dark:border-amber-800",
    88→        icon: <Loader2 className="size-5 animate-spin text-amber-600 dark:text-amber-400" />,
    89→        text: "Deployment is being provisioned…",
    90→      };
    91→    case "failed":
    92→    case "dead":
    93→      return {
    94→        bg: "bg-red-50 border-red-200 dark:bg-red-950/40 dark:border-red-800",
    95→        icon: <AlertCircle className="size-5 text-red-600 dark:text-red-400" />,
    96→        text: "Deployment failed. Try restarting or delete and re-deploy.",
    97→      };
    98→    case "stopped":
    99→    case "exited":
   100→      return {
   101→        bg: "bg-zinc-50 border-zinc-200 dark:bg-zinc-900/40 dark:border-zinc-700",
   102→        icon: <Square className="size-5 text-zinc-500 dark:text-zinc-400" />,
   103→        text: "Deployment is stopped. Start it to make it accessible.",
   104→      };
   105→    default:
   106→      return {
   107→        bg: "bg-muted border-border",
   108→        icon: <Circle className="size-5 text-muted-foreground" />,
   109→        text: "Unknown status.",
   110→      };
   111→  }
   112→}

   114→export function DeploymentView({ id }: { id: string }) {
   115→  const [dep, setDep] = useState<DeploymentItem | null>(null);
   116→  const [logs, setLogs] = useState<LogLine[]>([]);
   117→  const [busy, setBusy] = useState(false);
   118→  const [notFound, setNotFound] = useState(false);
   119→  const [volumeKeys, setVolumeKeys] = useState<string[] | null>(null);
   120→  const [volumeValues, setVolumeValues] = useState<Record<string, string>>({});
   121→  const [volumeLoading, setVolumeLoading] = useState(false);
   122→  const [labelEditing, setLabelEditing] = useState(false);
   123→  const [labelDraft, setLabelDraft] = useState("");
   124→  const [labelSaving, setLabelSaving] = useState(false);
   125→  const logRef = useRef<HTMLDivElement>(null);
   126→
   127→  const load = useCallback(async () => {
   128→    try {
   129→      const { deployment } = await api<{ deployment: DeploymentItem }>(`/api/deployments/${id}`);
   130→      setDep(deployment);
   131→    } catch (e) {
   132→      if (e instanceof ApiError && (e.status === 404 || e.code === "NOT_FOUND")) setNotFound(true);
   133→    }
   134→  }, [id]);
   135→
   136→  const loadLogs = useCallback(async () => {
   137→    try {
   138→      const { logs } = await api<{ logs: LogLine[] }>(`/api/deployments/${id}/logs?tail=200`);
   139→      setLogs(logs);
   140→    } catch {
   141→      /* ignore */
   142→    }
   143→  }, [id]);
   144→
   145→  const loadVolume = useCallback(async () => {
   146→    if (!dep) return;
   147→    setVolumeLoading(true);
   148→    try {
   149→      const res = await api<{ result: { ok: boolean; keys?: string[] } }>(`/api/deployments/${id}/volume`);
   150→      const keys = res.result.keys ?? [];
   151→      setVolumeKeys(keys);
   152→      // Fetch values for each key (limit to first 20 keys to avoid overload).
   153→      const limited = keys.slice(0, 20);
   154→      const entries = await Promise.all(
   155→        limited.map(async (k) => {
   156→          try {
   157→            const r = await api<{ result: { ok: boolean; value?: string | null } }>(
   158→              `/api/deployments/${id}/volume?op=get&key=${encodeURIComponent(k)}`
   159→            );
   160→            return [k, r.result.value ?? ""] as const;
   161→          } catch {
   162→            return [k, ""] as const;
   163→          }
   164→        })
   165→      );
   166→      setVolumeValues(Object.fromEntries(entries));
   167→    } catch {
   168→      setVolumeKeys([]);
   169→    } finally {
   170→      setVolumeLoading(false);
   171→    }
   172→  }, [id, dep]);
   173→
   174→  useEffect(() => {
   175→    load();
   176→    loadLogs();
   177→    const t = setInterval(() => {
   178→      load();
   179→      loadLogs();
   180→    }, 3000);
   181→    return () => clearInterval(t);
   182→  }, [load, loadLogs]);
   183→
   184→  useEffect(() => {
   185→    if (logRef.current) logRef.current.scrollTop = logRef.current.scrollHeight;
   186→  }, [logs]);
   187→
   188→  async function act(action: "start" | "stop" | "restart") {
   189→    setBusy(true);
   190→    try {
   191→      const { deployment } = await api<{ deployment: DeploymentItem }>(`/api/deployments/${id}/${action}`, {
   192→        method: "POST",
   193→      });
   194→      setDep(deployment);
   195→      toast.success(`Deployment ${action === 'stop' ? 'stopped' : action + 'ed'}`);
   196→      setTimeout(loadLogs, 400);
   197→    } catch (e) {
   198→      toast.error(e instanceof ApiError ? e.message : `Failed to ${action}`);
   199→    } finally {
   200→      setBusy(false);
   201→    }
   202→  }
   203→
   204→  async function remove() {
   205→    setBusy(true);
   206→    try {
   207→      await api(`/api/deployments/${id}`, { method: "DELETE" });
   208→      toast.success("Deployment deleted");
   209→      navigate({ name: "dashboard" });
   210→    } catch (e) {
   211→      toast.error(e instanceof ApiError ? e.message : "Failed to delete");
   212→      setBusy(false);
   213→    }
   214→  }
   215→
   216→  function startEditLabel() {
   217→    setLabelDraft(dep?.label ?? "");

   219→  }
   220→

   225→        method: "PATCH",
   226→        body: JSON.stringify({ label: labelDraft }),

   229→      setLabelEditing(false);

   231→    } catch (e) {
   232→      toast.error(e instanceof ApiError ? e.message : "Failed to update label");

   234→      setLabelSaving(false);
   235→    }

   238→  if (notFound) {

   242→        <p className="mt-2 text-sm text-muted-foreground">It may have been deleted or belongs to another account.</p>

   253→        <div className="h-6 w-40 animate-pulse rounded bg-muted" />

   255→      </div>
   256→    );
   257→  }
   258→
   259→  const running = dep.status === "running";
   260→  const steps = computeSteps(dep.status);

   262→  const containerId = dep.containerId ?? dep.id;
   263→  const metrics = getContainerMetrics(containerId);
   264→  const uptime = calculateUptime(dep.createdAt, dep.status);
   265→  const healthCheckTime = lastHealthCheck(containerId);

   268→    <div className="mx-auto max-w-4xl px-4 py-8">
   269→      <button
   270→        onClick={() => navigate({ name: "dashboard" })}

   273→        <ArrowLeft className="size-4" /> Dashboard

   275→
   276→      {/* Status banner */}

   278→        {banner.icon}
   279→        <div>
   280→          <p className="text-sm font-medium">{dep.status.charAt(0).toUpperCase() + dep.status.slice(1)}</p>
   281→          <p className="text-xs text-muted-foreground">{banner.text}</p>
   282→        </div>
   283→      </div>
   284→
   285→      {/* Health metrics section */}
   286→      <div className="mb-6 rounded-2xl border border-border bg-card p-5 shadow-sm">

   301→              />

   307→            icon={<MemoryStick className="size-3.5" />}

   310→          >
   311→            <Progress
   312→              value={metrics.memoryUsagePercent}
   313→              className="mt-2 h-1.5"
   314→            />
   315→          </MetricCard>
   316→
   317→          {/* CPU usage */}
   318→          <MetricCard
   319→            icon={<Cpu className="size-3.5" />}
   320→            label="CPU"
   321→            value={`${metrics.cpuUsagePercent}%`}
   322→          >
   323→            <Progress
   324→              value={metrics.cpuUsagePercent}
   325→              className="mt-2 h-1.5"
   326→            />

   328→
   329→          {/* Response latency */}

   333→            value={`${metrics.responseLatencyMs}ms`}

   335→            <div className="mt-2 flex items-center gap-1.5">
   336→              <div
   337→                className={`h-1.5 rounded-full ${
   338→                  metrics.responseLatencyMs < 50

   340→                    : metrics.responseLatencyMs < 100
   341→                    ? "bg-amber-500"
   342→                    : "bg-red-500"
   343→                }`}
   344→                style={{ width: `${Math.min((metrics.responseLatencyMs / 150) * 100, 100)}%` }}
   345→              />
   346→              <div className="h-1.5 flex-1 rounded-full bg-muted" />
   347→            </div>
   348→          </MetricCard>
   349→
   350→          {/* Last health check */}
   351→          <MetricCard
   352→            icon={<Clock className="size-3.5" />}
   353→            label="Last check"
   354→            value={formatHealthCheckTime(healthCheckTime)}
   355→          >
   356→            <p className="mt-1 text-[10px] text-muted-foreground">
   357→              {healthCheckTime.toLocaleTimeString()}
   358→            </p>
   359→          </MetricCard>
   360→        </div>
   361→      </div>
   362→

   364→      <DeploymentPerformanceSparklines
   365→        containerId={containerId}
   366→        status={dep.status}
   367→        metrics={metrics}
   368→      />
   369→
   370→      {/* Header */}
   371→      <div className="gradient-border-animated flex flex-col gap-4 rounded-2xl border border-border bg-card p-5 shadow-sm">
   372→        <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   373→          <div className="flex items-center gap-3">
   374→            <AppLogo logo={dep.app?.logo ?? null} simulator={dep.app?.simulator ?? "static"} name={dep.app?.name ?? "App"} size="lg" />
   375→            <div className="min-w-0">
   376→              <div className="flex flex-wrap items-center gap-2">
   377→                <h1 className="text-xl font-bold tracking-tight">{dep.app?.name ?? "Deployment"}</h1>
   378→                <span className={`inline-flex items-center gap-2 rounded-full px-2.5 py-0.5 text-[11px] font-medium ${statusColor(dep.status)}`}>
   379→                  <span className={`size-1.5 rounded-full ${statusDot(dep.status)}`} />
   380→                  {dep.status}
   381→                </span>

   383→                  <Timer className="size-3" />

   387→              <a

   391→                className="mt-1 inline-flex max-w-full items-center gap-1 font-mono text-xs text-brand hover:underline"

   395→              {/* Label row */}

   408→                      className="h-7 w-64 text-xs"
   409→                      autoFocus
   410→                      disabled={labelSaving}
   411→                    />
   412→                    <button
   413→                      onClick={saveLabel}
   414→                      disabled={labelSaving}
   415→                      className="inline-flex size-7 items-center justify-center rounded-md bg-brand text-brand-foreground hover:bg-brand/90 disabled:opacity-50"
   416→                      aria-label="Save label"
   417→                    >
   418→                      {labelSaving ? <Loader2 className="size-3.5 animate-spin" /> : <Check className="size-3.5" />}

   420→                    <button

   432→                  >
   433→                    <Tag className="size-3" />
   434→                    {dep.label ? (
   435→                      <span className="text-foreground">{dep.label}</span>
   436→                    ) : (

   438→                    )}
   439→                    <Pencil className="size-2.5 opacity-60" />
   440→                  </button>

   442→              </div>
   443→            </div>

   445→        </div>

   447→        <div className="flex flex-wrap items-center gap-1.5 border-t border-border/60 pt-3">

   451→            onClick={() => window.open(dep.previewPath, "_blank")}
   452→            disabled={!running}
   453→          >

   483→                <AlertDialogCancel>Cancel</AlertDialogCancel>

   494→      </div>
   495→
   496→      {/* Status Badge section */}
   497→      <StatusBadgeSection
   498→        appName={dep.app?.name ?? "app"}
   499→        status={dep.status}

   502→
   503→      {/* Deploy progress — connected line/dots visual */}
   504→      <div className="mt-6 rounded-2xl border border-border bg-card p-5 shadow-sm">
   505→        <h2 className="text-sm font-semibold">Deployment progress</h2>
   506→        <div className="relative mt-4">
   507→          {/* Vertical connecting line */}
   508→          <div className="absolute left-[11px] top-2 bottom-2 w-0.5 bg-border" />
   509→          <ol className="relative space-y-4">

   531→                >

   777→function computeSteps(status: string): { label: string; state: "done" | "active" | "pending" }[] {
   778→  const labels = [
   779→    "Creating deployment",
   780→    "Creating volume",
   781→    "Creating container",
   782→    "Starting application",
   783→    "Deployment ready",
   784→  ];
   785→  const order: Record<string, number> = {
   786→    pending: 0,
   787→    creating: 1,
   788→    created: 2,
   789→    running: 5,
   790→    stopped: 5,
   791→    exited: 5,
   792→    failed: 5,
   793→    dead: 5,
   794→  };
   795→  const progress = order[status] ?? 0;
   796→  return labels.map((label, i) => {
   797→    if (progress >= 5) return { label, state: "done" as const };
   798→    if (i < progress) return { label, state: "done" as const };
   799→    if (i === progress) return { label, state: "active" as const };
   800→    return { label, state: "pending" as const };
   801→  });
   802→}

   804→function timeAgo(iso: string): string {
   805→  const diff = Date.now() - new Date(iso).getTime();
   806→  const mins = Math.floor(diff / 60000);
   807→  if (mins < 1) return "just now";
   808→  if (mins < 60) return `${mins}m ago`;
   809→  const hrs = Math.floor(mins / 60);
   810→  if (hrs < 24) return `${hrs}h ago`;
   811→  const days = Math.floor(hrs / 24);
   812→  return `${days}d ago`;
   813→}

   815→function uptimeSince(iso: string): string {
   816→  const diff = Date.now() - new Date(iso).getTime();
   817→  const mins = Math.floor(diff / 60000);
   818→  if (mins < 1) return "<1m";
   819→  if (mins < 60) return `${mins}m`;
   820→  const hrs = Math.floor(mins / 60);
   821→  const remainMins = mins % 60;
   822→  if (hrs < 24) return remainMins > 0 ? `${hrs}h ${remainMins}m` : `${hrs}h`;
   823→  const days = Math.floor(hrs / 24);
   824→  return `${days}d`;
   825→}

   847→  const baseDisk = metrics.healthScore * 0.3 + 5; // ~5-35
   848→

   850→  const [tick, setTick] = useState(0);
   851→
   852→  // Auto-refresh every 5 seconds for running deployments
   853→  useEffect(() => {
   854→    if (!running) return;

   871→      d = tickTimeSeries(d, baseDisk, 3, i + 1);

   874→  }, [containerId, baseCpu, baseMemory, baseNetwork, baseDisk, tick]);
