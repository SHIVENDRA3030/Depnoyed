    23→function getFavorites(): string[] {
    24→  if (typeof window === "undefined") return [];
    25→  try { return JSON.parse(localStorage.getItem("oss-favorites") || "[]"); } catch { return []; }
    26→}
    27→
    28→function getRecentSearches(): string[] {
    29→  if (typeof window === "undefined") return [];
    30→  try { return JSON.parse(localStorage.getItem("oss-recent-searches") || "[]"); } catch { return []; }
    31→}
    32→
    33→function addRecentSearch(q: string) {
    34→  if (!q.trim()) return;
    35→  try {
    36→    const existing = getRecentSearches().filter(s => s !== q);
    37→    const next = [q, ...existing].slice(0, 5);
    38→    localStorage.setItem("oss-recent-searches", JSON.stringify(next));
    39→  } catch {}
    40→}
    41→
    42→function clearRecentSearches() {
    43→  try { localStorage.removeItem("oss-recent-searches"); } catch {}
    44→}

   411→function AppCard({ app, onFavToggle }: { app: AppItem; onFavToggle?: () => void }) {
   412→  const user = useAuth((s) => s.user);
   413→  const [modalOpen, setModalOpen] = useState(false);
   414→  const inCompare = useCompare((s) => s.has(app.slug));
   415→  const toggleCompare = useCompare((s) => s.toggle);
   416→  const compareCount = useCompare((s) => s.slugs.length);
   417→  const [fav, setFav] = useState(() => {
   418→    if (typeof window === "undefined") return false;
   419→    try {
   420→      return JSON.parse(localStorage.getItem("oss-favorites") || "[]").includes(app.slug);
   421→    } catch { return false; }
   422→  });
   423→
   424→  function handleDeploy(e: React.MouseEvent) {
   425→    e.stopPropagation();
   426→    e.preventDefault();
   427→    if (!user) {
   428→      toast.info("Sign in to deploy");
   429→      navigate({ name: "login" });
   430→      return;
   431→    }
   432→    setModalOpen(true);
   433→  }
   434→
   435→  function toggleFav(e: React.MouseEvent) {
   436→    e.stopPropagation();
   437→    e.preventDefault();
   438→    try {
   439→      const favs: string[] = JSON.parse(localStorage.getItem("oss-favorites") || "[]");
   440→      const next = favs.includes(app.slug) ? favs.filter(f => f !== app.slug) : [...favs, app.slug];
   441→      localStorage.setItem("oss-favorites", JSON.stringify(next));
   442→      setFav(!fav);
   443→      onFavToggle?.();
   444→    } catch {}
   445→  }
   446→
   447→  return (
   448→    <div
