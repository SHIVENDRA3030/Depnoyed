    24→    24→
    25→    25→function AnimatedCounter({ target, duration = 1200, className = "" }: { target: number; duration?: number; className?: string }) {
    26→    26→  const [display, setDisplay] = useState(0);
    27→    27→  const ref = useRef<{ start: number | null; raf: number }>({ start: null, raf: 0 });
    28→    28→
    29→    29→  useEffect(() => {
    30→    30→    const startTime = performance.now();
    31→    31→    ref.current.start = startTime;
    32→    32→
    33→    33→    function step(now: number) {
    34→    34→      const elapsed = now - startTime;
    35→    35→      const progress = Math.min(elapsed / duration, 1);
    36→    36→      // ease-out cubic for a nice deceleration
    37→    37→      const eased = 1 - Math.pow(1 - progress, 3);
    38→    38→      setDisplay(Math.round(eased * target));
    39→    39→      if (progress < 1) {
    40→    40→        ref.current.raf = requestAnimationFrame(step);
    41→    41→      }
    42→    42→    }
    43→    43→
    44→    44→    ref.current.raf = requestAnimationFrame(step);
    45→    45→    return () => cancelAnimationFrame(ref.current.raf);
    46→    46→  }, [target, duration]);
    47→    47→
    48→    48→  return <span className={className}>{display}</span>;
    49→    49→}

    75→    75→  if (!q.trim()) return;

    77→    77→    const existing = getRecentSearches().filter(s => s !== q);

    79→    79→    localStorage.setItem("oss-recent-searches", JSON.stringify(next));
    80→    80→  } catch {}
    81→    81→}
