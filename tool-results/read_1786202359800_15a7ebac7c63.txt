    74→function statusAccentBorder(status: string): string {
    75→  switch (status) {
    76→    case "running":
    77→      return "border-l-emerald-500";
    78→    case "pending":
    79→    case "creating":
    80→      return "border-l-amber-500";
    81→    case "failed":
    82→    case "dead":
    83→      return "border-l-red-500";
    84→    case "stopped":
    85→    case "exited":
    86→      return "border-l-zinc-400 dark:border-l-zinc-500";
    87→    default:
    88→      return "border-l-muted-foreground";
    89→  }
    90→}
    91→
    92→function statusDotClass(status: string): string {
    93→  switch (status) {
    94→    case "running":
    95→      return "bg-emerald-500";
    96→    case "stopped":
    97→    case "exited":
    98→      return "bg-zinc-400";
    99→    case "pending":
   100→    case "creating":
   101→      return "bg-amber-500";
   102→    case "failed":
   103→    case "dead":
   104→      return "bg-red-500";
   105→    default:
   106→      return "bg-muted-foreground";
   107→  }

   109→
   110→export function DashboardView() {
   111→  const [deployments, setDeployments] = useState<DeploymentItem[] | null>(null);
   112→  const [busyId, setBusyId] = useState<string | null>(null);
   113→  const [refreshing, setRefreshing] = useState(false);
   114→  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set());
   115→  const [batchBusy, setBatchBusy] = useState(false);
   116→  const [searchQuery, setSearchQuery] = useState("");
   117→  const [statusFilter, setStatusFilter] = useState<string>("all");
   118→
   119→  const load = useCallback(async () => {
   120→    try {
   121→      const { deployments } = await api<{ deployments: DeploymentItem[] }>("/api/deployments");
   122→      setDeployments(deployments);
   123→    } catch {
   124→      toast.error("Failed to load deployments");
   125→      setDeployments([]);
   126→    }
   127→  }, []);
   128→
   129→  useEffect(() => {
   130→    load();
   131→    const t = setInterval(load, 5000);
   132→    return () => clearInterval(t);
   133→  }, [load]);
   134→
   135→  const filteredDeployments = useMemo(() => {
   136→    if (!deployments) return [];
   137→    const q = searchQuery.trim().toLowerCase();
   138→    return deployments.filter((d) => {
   139→      const matchesStatus =
   140→        statusFilter === "all" ||
   141→        (statusFilter === "running" && d.status === "running") ||
   142→        (statusFilter === "stopped" && (d.status === "stopped" || d.status === "exited"));
   143→      if (!matchesStatus) return false;
   144→      if (!q) return true;
   145→      return (
   146→        (d.app?.name?.toLowerCase().includes(q) ?? false) ||
   147→        (d.label?.toLowerCase().includes(q) ?? false) ||
   148→        d.subdomain.toLowerCase().includes(q) ||
   149→        d.containerName.toLowerCase().includes(q)
   150→      );
   151→    });
   152→  }, [deployments, searchQuery, statusFilter]);
   153→
   154→  async function refresh() {
   155→    setRefreshing(true);
   156→    await load();
   157→    setRefreshing(false);
   158→  }
   159→
   160→  async function act(id: string, action: "start" | "stop" | "restart") {
   161→    setBusyId(id);
   162→    try {
   163→      const { deployment } = await api<{ deployment: DeploymentItem }>(`/api/deployments/${id}/${action}`, {
   164→        method: "POST",
   165→      });
   166→      setDeployments((d) => (d ?? []).map((x) => (x.id === id ? deployment : x)));
   167→      toast.success(`Deployment ${action === 'stop' ? 'stopped' : action + 'ed'}`);
   168→    } catch (e) {
   169→      toast.error(e instanceof ApiError ? e.message : `Failed to ${action}`);
   170→    } finally {
   171→      setBusyId(null);
   172→    }
   173→  }
   174→
   175→  async function remove(id: string) {
   176→    setBusyId(id);
   177→    try {
   178→      await api(`/api/deployments/${id}`, { method: "DELETE" });
   179→      setDeployments((d) => (d ?? []).filter((x) => x.id !== id));
   180→      toast.success("Deployment deleted");
   181→    } catch (e) {
   182→      toast.error(e instanceof ApiError ? e.message : "Failed to delete");
   183→    } finally {
   184→      setBusyId(null);
   185→    }
   186→  }
   187→
   188→  // Batch operations
   189→  function toggleSelect(id: string) {
   190→    setSelectedIds((prev) => {
   191→      const next = new Set(prev);
   192→      if (next.has(id)) next.delete(id);
   193→      else next.add(id);
   194→      return next;
   195→    });
   196→  }
   197→
   198→  function toggleSelectAll() {
   199→    if (!deployments) return;
   200→    if (selectedIds.size === deployments.length) {
   201→      setSelectedIds(new Set());
   202→    } else {
   203→      setSelectedIds(new Set(deployments.map((d) => d.id)));
   204→    }
   205→  }
   206→
   207→  function deselectAll() {
   208→    setSelectedIds(new Set());
   209→  }
   210→
   211→  const batchInfo = useMemo(() => {
   212→    if (!deployments) return { hasStopped: false, hasRunning: false, count: 0 };
   213→    const selected = deployments.filter((d) => selectedIds.has(d.id));
   214→    return {
   215→      hasStopped: selected.some((d) => d.status === "stopped" || d.status === "exited"),
   216→      hasRunning: selected.some((d) => d.status === "running"),
   217→      count: selected.length,
   218→    };
   219→  }, [deployments, selectedIds]);
   220→
   221→  async function batchAction(action: "start" | "stop" | "restart") {
   222→    setBatchBusy(true);
   223→    let successCount = 0;
   224→    let failCount = 0;
   225→    for (const id of selectedIds) {
   226→      try {
   227→        await api<{ deployment: DeploymentItem }>(`/api/deployments/${id}/${action}`, { method: "POST" });
   228→        successCount++;
   229→      } catch {
   230→        failCount++;
   231→      }
   232→    }
   233→    await load();
   234→    setBatchBusy(false);
   235→    deselectAll();
   236→    if (failCount === 0) toast.success(`${successCount} deployment(s) ${action === 'stop' ? 'stopped' : action + 'ed'}`);
   237→    else toast.error(`${failCount} failed, ${successCount} succeeded`);
   238→  }
   239→
   240→  async function batchDelete() {
   241→    setBatchBusy(true);
   242→    let successCount = 0;
   243→    let failCount = 0;
   244→    for (const id of selectedIds) {
   245→      try {
   246→        await api(`/api/deployments/${id}`, { method: "DELETE" });
   247→        successCount++;
   248→      } catch {
   249→        failCount++;
   250→      }
   251→    }
   252→    await load();
   253→    setBatchBusy(false);
   254→    deselectAll();
   255→    if (failCount === 0) toast.success(`${successCount} deployment(s) deleted`);
   256→    else toast.error(`${failCount} failed, ${successCount} succeeded`);
   257→  }
   258→
   259→  // Derived data for health monitoring
   260→  const healthData = useMemo(() => {
   261→    if (!deployments || deployments.length === 0) return null;
   262→    const runningDeps = deployments.filter((d) => d.status === "running");
   263→    const perDeployment = deployments.map((d) => {
   264→      const containerId = d.containerId ?? d.id;
   265→      const metrics = getContainerMetrics(containerId);
   266→      const uptime = calculateUptime(d.createdAt, d.status);
   267→      const lastCheck = lastHealthCheck(containerId);
   268→      return {
   269→        id: d.id,
   270→        name: d.app?.name ?? "Unknown",
   271→        status: d.status,
   272→        uptime,
   273→        healthScore: d.status === "running" ? Math.max(80, metrics.healthScore) : d.status === "stopped" || d.status === "exited" ? 0 : metrics.healthScore,
   274→        lastCheck,
   275→        responseTime: d.status === "running" ? metrics.responseLatencyMs : null,
   276→      };
   277→    });
   278→    // Overall health: average of running deployments' health scores
   279→    const overallHealth = runningDeps.length > 0
   280→      ? Math.round(runningDeps.reduce((acc, d) => {
   281→          const m = getContainerMetrics(d.containerId ?? d.id);
   282→          return acc + Math.max(80, m.healthScore);
   283→        }, 0) / runningDeps.length)
   284→      : 0;
   285→    // Simulated 24h uptime bars (hourly data)
   286→    const uptime24h = Array.from({ length: 24 }, (_, i) => {
   287→      const seed = `uptime24h-${i}-${runningDeps.length}`;
   288→      const r = seededRandom(seed);
   289→      if (runningDeps.length === 0) return 0;
   290→      return 80 + r * 20; // 80-100% per hour when running
   291→    });
   292→    return { overallHealth, perDeployment, uptime24h, runningCount: runningDeps.length };
   293→  }, [deployments]);
   294→
   295→  const [timelineExpanded, setTimelineExpanded] = useState(false);
   296→
   297→  return (
   298→    <div className="mx-auto max-w-6xl px-4 py-8">
   299→      {/* Header with gradient bg */}
   300→      <div className="dashboard-header-gradient -mx-4 -mt-8 px-4 pt-8 pb-6">
   301→        <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   302→          <div>
   303→            <h1 className="text-2xl font-bold tracking-tight">My deployments</h1>
   304→            <p className="text-sm text-muted-foreground">Manage your isolated application instances.</p>
   305→          </div>
   306→          <div className="flex items-center gap-2">
   307→            <Button variant="outline" size="sm" onClick={refresh} disabled={refreshing}>
   308→              {refreshing ? <Loader2 className="mr-2 size-4 animate-spin" /> : <RefreshCw className="mr-2 size-4" />}
   309→              Refresh
   310→            </Button>
   311→            <Button size="sm" className="bg-brand text-brand-foreground hover:bg-brand/90" onClick={() => navigate({ name: "marketplace" })}>
   312→              <Plus className="mr-2 size-4" /> Deploy new
   313→            </Button>
   314→          </div>
   315→        </div>
   316→      </div>
   317→
   318→      {/* Stats cards with glass morphism */}
   319→      {deployments && deployments.length > 0 && (
   320→        <div className="mt-6 grid grid-cols-2 gap-3 sm:grid-cols-4">
   321→          <StatCard
   322→            icon={<Boxes className="size-4" />}
   323→            label="Total"
   324→            value={String(deployments.length)}
   325→            tone="default"
   326→            glass
   327→          />
   328→          <StatCard
   329→            icon={<Activity className="size-4" />}
   330→            label="Running"
   331→            value={String(deployments.filter((d) => d.status === "running").length)}
   332→            tone="emerald"
   333→            glass
   334→          />
   335→          <StatCard
   336→            icon={<Square className="size-4" />}
   337→            label="Stopped"
   338→            value={String(deployments.filter((d) => d.status === "stopped" || d.status === "exited").length)}
   339→            tone="zinc"
   340→            glass
   341→          />
   342→          <StatCard
   343→            icon={<HardDrive className="size-4" />}
   344→            label="Data stored"
   345→            value={formatDataSize(
   346→              deployments.reduce((acc, d) => acc + (d.volumeDataSize ?? 0), 0)
   347→            )}
   348→            tone="brand"
   349→            glass
   350→          />
   351→        </div>
   352→      )}
   353→
   354→      {/* Health Overview — Feature 1 */}
   355→      {healthData && healthData.runningCount > 0 && (
   356→        <HealthOverview
   357→          overallHealth={healthData.overallHealth}
   358→          uptime24h={healthData.uptime24h}
   359→          perDeployment={healthData.perDeployment}
   360→        />
   361→      )}
   362→
   363→      {/* Quick Actions — Feature 2 */}
   364→      <QuickActions
   365→        hasDeployments={!!deployments && deployments.length > 0}
   366→        runningCount={deployments?.filter((d) => d.status === "running").length ?? 0}
   367→        onRefreshAll={async () => {
   368→          if (!deployments) return;
   369→          const running = deployments.filter((d) => d.status === "running");
   370→          for (const d of running) {
   371→            try { await api(`/api/deployments/${d.id}/restart`, { method: "POST" }); } catch { /* continue */ }
   372→          }
   373→          await load();
   374→          toast.success(`${running.length} deployment(s) refreshed`);
   375→        }}
   376→        onStopAll={async () => {
   377→          if (!deployments) return;
   378→          const running = deployments.filter((d) => d.status === "running");
   379→          for (const d of running) {
   380→            try { await api(`/api/deployments/${d.id}/stop`, { method: "POST" }); } catch { /* continue */ }
   381→          }
   382→          await load();
   383→          toast.success(`${running.length} deployment(s) stopped`);
   384→        }}
   385→      />
   386→
   387→      {/* Cost estimation */}
   388→      {deployments && deployments.length > 0 && (
   389→        <div className="mt-4 rounded-2xl border border-border bg-gradient-to-br from-card via-card to-emerald-50/30 p-5 shadow-sm dark:to-emerald-950/10">
   390→          <div className="flex items-center justify-between">
   391→            <div className="flex items-center gap-2">
   392→              <span className="flex size-8 items-center justify-center rounded-lg bg-brand-soft text-brand">
   393→                <DollarSign className="size-4" />
   394→              </span>
   395→              <div>
   396→                <h3 className="text-sm font-semibold">Estimated usage cost</h3>
   397→                <p className="text-[11px] text-muted-foreground">Based on running deployments × resource usage</p>
   398→              </div>
   399→            </div>
   400→            <div className="text-right">
   401→              <p className="text-lg font-bold">
   402→                ${estimateCost(deployments).monthly.toFixed(2)}
   403→                <span className="ml-1 text-xs font-normal text-muted-foreground">/mo</span>
   404→              </p>
   405→              <p className="text-[11px] text-muted-foreground">
   406→                ${estimateCost(deployments).daily.toFixed(3)}/day
   407→              </p>
   408→            </div>
   409→          </div>
   410→          <div className="mt-3 flex flex-wrap gap-3">
   411→            <div className="flex items-center gap-1.5 rounded-full border border-border/60 bg-background/60 px-2.5 py-1 text-xs">
   412→              <Cpu className="size-3 text-brand" />
   413→              <span className="text-muted-foreground">CPU:</span>
   414→              <span className="font-medium">${(0.5 * deployments.filter((d) => d.status === "running").length * 5).toFixed(2)}/mo</span>
   415→            </div>
   416→            <div className="flex items-center gap-1.5 rounded-full border border-border/60 bg-background/60 px-2.5 py-1 text-xs">
   417→              <MemoryStick className="size-3 text-brand" />
   418→              <span className="text-muted-foreground">Memory:</span>
   419→              <span className="font-medium">${(0.5 * deployments.filter((d) => d.status === "running").length * 3).toFixed(2)}/mo</span>
   420→            </div>
   421→            <div className="flex items-center gap-1.5 rounded-full border border-border/60 bg-background/60 px-2.5 py-1 text-xs">
   422→              <HardDrive className="size-3 text-brand" />
   423→              <span className="text-muted-foreground">Storage:</span>
   424→              <span className="font-medium">${(Math.max(1, deployments.reduce((a, d) => a + (d.volumeDataSize ?? 0), 0) / 1024 / 1024) * 0.10).toFixed(2)}/mo</span>
   425→            </div>
   426→          </div>
   427→          <p className="mt-3 text-[10px] text-muted-foreground/70">
   428→            Estimation only — actual costs may vary based on provider pricing and usage patterns.
   429→          </p>
   430→        </div>
   431→      )}
   432→
   433→      {/* Search & filter bar */}
   434→      {deployments && deployments.length > 0 && (
   435→        <div className="mt-4 flex flex-col gap-2 sm:flex-row sm:items-center">
   436→          <div className="relative flex-1">
   437→            <Search className="pointer-events-none absolute left-3 top-1/2 size-4 -translate-y-1/2 text-muted-foreground" />
   438→            <Input
   439→              value={searchQuery}
   440→              onChange={(e) => setSearchQuery(e.target.value)}
   441→              placeholder="Search by app name, label, or subdomain…"
   442→              className="pl-9"
   443→            />
   444→            {searchQuery && (
   445→              <button
   446→                onClick={() => setSearchQuery("")}
   447→                className="absolute right-2 top-1/2 -translate-y-1/2 rounded p-1 text-muted-foreground hover:bg-muted hover:text-foreground"
   448→              >
   449→                <X className="size-3.5" />
   450→              </button>
   451→            )}
   452→          </div>
   453→          <div className="flex items-center gap-1.5">
   454→            {["all", "running", "stopped"].map((f) => (
   455→              <button
   456→                key={f}
   457→                onClick={() => setStatusFilter(f)}
   458→                className={`inline-flex items-center gap-1.5 rounded-full border px-3 py-1.5 text-xs font-medium transition-all ${
   459→                  statusFilter === f
   460→                    ? "border-brand/40 bg-brand/10 text-brand"
   461→                    : "border-border bg-background text-muted-foreground hover:bg-muted"
   462→                }`}
   463→              >
   464→                {f === "all" && <Boxes className="size-3" />}
   465→                {f === "running" && <Activity className="size-3" />}
   466→                {f === "stopped" && <Square className="size-3" />}
   467→                {f.charAt(0).toUpperCase() + f.slice(1)}
   468→                <span className="ml-0.5 text-[10px] text-muted-foreground/70">
   469→                  {f === "all"
   470→                    ? deployments.length
   471→                    : f === "running"
   472→                    ? deployments.filter((d) => d.status === "running").length
   473→                    : deployments.filter((d) => d.status === "stopped" || d.status === "exited").length}
   474→                </span>
   475→              </button>
   476→            ))}
   477→          </div>
   478→        </div>
   479→      )}
   480→
   481→      <div className="mt-6">
   482→        {deployments === null ? (
   483→          <div className="space-y-3">
   484→            {Array.from({ length: 2 }).map((_, i) => (
   485→              <div key={i} className="h-28 rounded-2xl shimmer" />
   486→            ))}
   487→          </div>
   488→        ) : deployments.length === 0 ? (
   489→          <EmptyState />
   490→        ) : filteredDeployments.length === 0 ? (
   491→          <div className="flex flex-col items-center rounded-2xl border border-dashed border-border py-12 text-center">
   492→            <Search className="size-8 text-muted-foreground/40" />
   493→            <p className="mt-2 text-sm text-muted-foreground">No deployments match your search.</p>
   494→            <Button variant="outline" size="sm" className="mt-3" onClick={() => { setSearchQuery(""); setStatusFilter("all"); }}>
   495→              Clear filters
   496→            </Button>
   497→          </div>
   498→        ) : (
   499→          <div className="space-y-3">
   500→            {filteredDeployments.map((d) => (
   501→              <DeploymentRow
   502→                key={d.id}
   503→                d={d}
   504→                busy={busyId === d.id}
   505→                selected={selectedIds.has(d.id)}
   506→                onSelect={() => toggleSelect(d.id)}
   507→                onStart={() => act(d.id, "start")}
   508→                onStop={() => act(d.id, "stop")}
   509→                onRestart={() => act(d.id, "restart")}
   510→                onDelete={() => remove(d.id)}
   511→              />
   512→            ))}
   513→          </div>
   514→        )}
   515→      </div>
   516→
   517→      {/* Resource Usage sparkline section */}
   518→      {deployments && deployments.length > 0 && (
   519→        <ResourceUsageSparklines deployments={deployments} />
   520→      )}
   521→
   522→      {/* Activity timeline - Enhanced (Feature 3) */}
   523→      {deployments && deployments.length > 0 && (
   524→        <EnhancedActivityTimeline
   525→          deployments={deployments}
   526→          expanded={timelineExpanded}
   527→          onToggleExpand={() => setTimelineExpanded((v) => !v)}
   528→        />
   529→      )}
   530→
   531→      {/* Batch action bar - fixed at bottom */}
   532→      {batchInfo.count > 0 && (
   533→        <div className="fixed inset-x-0 bottom-0 z-50 border-t border-border bg-card/95 shadow-lg backdrop-blur-sm animate-in slide-in-from-bottom-4 duration-200">
   534→          <div className="mx-auto flex max-w-6xl items-center justify-between gap-3 px-4 py-3">
   535→            <div className="flex items-center gap-3">
   536→              <Checkbox
   537→                checked={deployments ? selectedIds.size === deployments.length : false}
   538→                onCheckedChange={toggleSelectAll}
   539→                aria-label="Select all"

   868→function timeAgo(iso: string): string {
   869→  const diff = Date.now() - new Date(iso).getTime();
   870→  const mins = Math.floor(diff / 60000);
   871→  if (mins < 1) return "just now";
   872→  if (mins < 60) return `${mins}m ago`;
   873→  const hrs = Math.floor(mins / 60);
   874→  if (hrs < 24) return `${hrs}h ago`;
   875→  const days = Math.floor(hrs / 24);
   876→  return `${days}d ago`;
   877→}
   878→
   879→function uptimeSince(iso: string): string {
   880→  const diff = Date.now() - new Date(iso).getTime();
   881→  const mins = Math.floor(diff / 60000);
   882→  if (mins < 1) return "<1m";
   883→  if (mins < 60) return `${mins}m`;
   884→  const hrs = Math.floor(mins / 60);
   885→  const remainMins = mins % 60;
   886→  if (hrs < 24) return remainMins > 0 ? `${hrs}h ${remainMins}m` : `${hrs}h`;
   887→  const days = Math.floor(hrs / 24);
   888→  return `${days}d`;
   889→}

   891→function formatDataSize(bytes: number): string {
   892→  if (bytes < 1024) return `${bytes} B`;
   893→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
   894→  return `${(bytes / (1024 * 1024)).toFixed(1)} MB`;
   895→}

   897→function estimateCost(deployments: DeploymentItem[]) {
   898→  const running = deployments.filter((d) => d.status === "running").length;
   899→  const cpuCost = 0.5 * running * 5;  // $5/core/month
   900→  const memCost = 0.5 * running * 3;   // $3/512MB/month
   901→  const totalStorage = deployments.reduce((a, d) => a + (d.volumeDataSize ?? 0), 0);
   902→  const storageGB = Math.max(1, totalStorage / 1024 / 1024);
   903→  const storageCost = storageGB * 0.10; // $0.10/GB/month
   904→  const monthly = cpuCost + memCost + storageCost;
   905→  const daily = monthly / 30;
   906→  return { monthly, daily };
   907→}
