    76→    76→function statusBannerStyle(status: string): { bg: string; icon: React.ReactNode; text: string } {
    77→    77→  switch (status) {
    78→    78→    case "running":
    79→    79→      return {
    80→    80→        bg: "bg-emerald-50 border-emerald-200 dark:bg-emerald-950/40 dark:border-emerald-800",
    81→    81→        icon: <CheckCircle2 className="size-5 text-emerald-600 dark:text-emerald-400" />,
    82→    82→        text: "This deployment is running and accessible.",
    83→    83→      };
    84→    84→    case "pending":
    85→    85→    case "creating":
    86→    86→      return {
    87→    87→        bg: "bg-amber-50 border-amber-200 dark:bg-amber-950/40 dark:border-amber-800",
    88→    88→        icon: <Loader2 className="size-5 animate-spin text-amber-600 dark:text-amber-400" />,
    89→    89→        text: "Deployment is being provisioned…",
    90→    90→      };
    91→    91→    case "failed":
    92→    92→    case "dead":
    93→    93→      return {
    94→    94→        bg: "bg-red-50 border-red-200 dark:bg-red-950/40 dark:border-red-800",
    95→    95→        icon: <AlertCircle className="size-5 text-red-600 dark:text-red-400" />,
    96→    96→        text: "Deployment failed. Try restarting or delete and re-deploy.",
    97→    97→      };
    98→    98→    case "stopped":
    99→    99→    case "exited":
   100→   100→      return {
   101→   101→        bg: "bg-zinc-50 border-zinc-200 dark:bg-zinc-900/40 dark:border-zinc-700",
   102→   102→        icon: <Square className="size-5 text-zinc-500 dark:text-zinc-400" />,
   103→   103→        text: "Deployment is stopped. Start it to make it accessible.",
   104→   104→      };
   105→   105→    default:
   106→   106→      return {
   107→   107→        bg: "bg-muted border-border",
   108→   108→        icon: <Circle className="size-5 text-muted-foreground" />,
   109→   109→        text: "Unknown status.",

   777→   777→function computeSteps(status: string): { label: string; state: "done" | "active" | "pending" }[] {
   778→   778→  const labels = [
   779→   779→    "Creating deployment",
   780→   780→    "Creating volume",
   781→   781→    "Creating container",
   782→   782→    "Starting application",
   783→   783→    "Deployment ready",
   784→   784→  ];
   785→   785→  const order: Record<string, number> = {
   786→   786→    pending: 0,
   787→   787→    creating: 1,
   788→   788→    created: 2,
   789→   789→    running: 5,
   790→   790→    stopped: 5,
   791→   791→    exited: 5,
   792→   792→    failed: 5,
   793→   793→    dead: 5,
   794→   794→  };
   795→   795→  const progress = order[status] ?? 0;
   796→   796→  return labels.map((label, i) => {
   797→   797→    if (progress >= 5) return { label, state: "done" as const };
   798→   798→    if (i < progress) return { label, state: "done" as const };
   799→   799→    if (i === progress) return { label, state: "active" as const };
   800→   800→    return { label, state: "pending" as const };
   801→   801→  });
   802→   802→}

   818→   818→  if (mins < 1) return "<1m";
   819→   819→  if (mins < 60) return `${mins}m`;
   820→   820→  const hrs = Math.floor(mins / 60);

   824→   824→  return `${days}d`;

   829→   829→const DEPLOY_SPARKLINE_POINTS = 30;

   831→   831→function DeploymentPerformanceSparklines({
   832→   832→  containerId,
   833→   833→  status,
   834→   834→  metrics,
   835→   835→}: {
   836→   836→  containerId: string;
   837→   837→  status: string;
   838→   838→  metrics: { cpuUsagePercent: number; memoryUsagePercent: number; responseLatencyMs: number; healthScore: number };
   839→   839→}) {
   840→   840→  const running = status === "running";
   841→   841→  const color = sparklineColor(status);
   842→   842→
   843→   843→  // Base metrics from getContainerMetrics
   844→   844→  const baseCpu = metrics.cpuUsagePercent;
   845→   845→  const baseMemory = metrics.memoryUsagePercent;
   846→   846→  const baseNetwork = metrics.responseLatencyMs / 150 * 40 + 10; // ~10-50
   847→   847→  const baseDisk = metrics.healthScore * 0.3 + 5; // ~5-35
   848→   848→
   849→   849→  // Tick counter drives series evolution (only ticks for running deployments)
   850→   850→  const [tick, setTick] = useState(0);
   851→   851→
   852→   852→  // Auto-refresh every 5 seconds for running deployments
   853→   853→  useEffect(() => {
   854→   854→    if (!running) return;
   855→   855→    const t = setInterval(() => setTick((n) => n + 1), 5000);
   856→   856→    return () => clearInterval(t);
   857→   857→  }, [running]);
   858→   858→
   859→   859→  // Compute full series from containerId + tick (fully derived)
   860→   860→  const series = useMemo(() => {
   861→   861→    const cpu = generateTimeSeries(baseCpu, DEPLOY_SPARKLINE_POINTS, 4, `dep-${containerId}:cpu`);
   862→   862→    const memory = generateTimeSeries(baseMemory, DEPLOY_SPARKLINE_POINTS, 3, `dep-${containerId}:mem`);
   863→   863→    const network = generateTimeSeries(baseNetwork, DEPLOY_SPARKLINE_POINTS, 5, `dep-${containerId}:net`);
   864→   864→    const disk = generateTimeSeries(baseDisk, DEPLOY_SPARKLINE_POINTS, 3, `dep-${containerId}:disk`);
   865→   865→    // Apply tick-based evolution
   866→   866→    let c = cpu, m = memory, n = network, d = disk;
   867→   867→    for (let i = 0; i < tick; i++) {
   868→   868→      c = tickTimeSeries(c, baseCpu, 4, i + 1);
   869→   869→      m = tickTimeSeries(m, baseMemory, 3, i + 1);
   870→   870→      n = tickTimeSeries(n, baseNetwork, 5, i + 1);
   871→   871→      d = tickTimeSeries(d, baseDisk, 3, i + 1);
   872→   872→    }
   873→   873→    return { cpu: c, memory: m, network: n, disk: d };
   874→   874→  }, [containerId, baseCpu, baseMemory, baseNetwork, baseDisk, tick]);
