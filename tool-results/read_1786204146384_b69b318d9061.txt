    21→type SortMode = "popular" | "newest" | "alpha" | "favorites";
    22→
    23→function getFavorites(): string[] {
    24→  if (typeof window === "undefined") return [];
    25→  try { return JSON.parse(localStorage.getItem("oss-favorites") || "[]"); } catch { return []; }
    26→}
    27→
    28→function getRecentSearches(): string[] {
    29→  if (typeof window === "undefined") return [];
    30→  try { return JSON.parse(localStorage.getItem("oss-recent-searches") || "[]"); } catch { return []; }
    31→}
    32→
    33→function addRecentSearch(q: string) {
    34→  if (!q.trim()) return;
    35→  try {
    36→    const existing = getRecentSearches().filter(s => s !== q);
    37→    const next = [q, ...existing].slice(0, 5);
    38→    localStorage.setItem("oss-recent-searches", JSON.stringify(next));
    39→  } catch {}
    40→}
    41→
    42→function clearRecentSearches() {
    43→  try { localStorage.removeItem("oss-recent-searches"); } catch {}
    44→}

   417→function AppCard({ app, onFavToggle }: { app: AppItem; onFavToggle?: () => void }) {
   418→  const user = useAuth((s) => s.user);
   419→  const [modalOpen, setModalOpen] = useState(false);
   420→  const inCompare = useCompare((s) => s.has(app.slug));
   421→  const toggleCompare = useCompare((s) => s.toggle);
   422→  const compareCount = useCompare((s) => s.slugs.length);
   423→  const [fav, setFav] = useState(() => {
   424→    if (typeof window === "undefined") return false;
   425→    try {
   426→      return JSON.parse(localStorage.getItem("oss-favorites") || "[]").includes(app.slug);
   427→    } catch { return false; }
   428→  });
   429→
