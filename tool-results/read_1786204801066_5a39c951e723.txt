   164→   164→function ProfileSection({
   165→   165→  user,

   169→   169→  hydrate: () => Promise<void>;

   175→   175→  async function saveProfile() {
   176→   176→    setBusy(true);
   177→   177→    try {
   178→   178→      await api("/api/auth/profile", {
   179→   179→        method: "PATCH",
   180→   180→        body: JSON.stringify({ name }),
   181→   181→      });

   186→   186→      toast.error(err instanceof ApiError ? err.message : "Failed to update profile");

   189→   189→    }

   246→   246→function SecuritySection() {
   247→   247→  const [changing, setChanging] = useState(false);
   248→   248→  const [currentPw, setCurrentPw] = useState("");
   249→   249→  const [newPw, setNewPw] = useState("");
   250→   250→  const [confirmPw, setConfirmPw] = useState("");
   251→   251→  const [busy, setBusy] = useState(false);
   252→   252→
   253→   253→  async function changePassword() {
   254→   254→    if (newPw !== confirmPw) {
   255→   255→      toast.error("Passwords don't match");
   256→   256→      return;
   257→   257→    }
   258→   258→    if (newPw.length < 6) {
   259→   259→      toast.error("Password must be at least 6 characters");
   260→   260→      return;
   261→   261→    }
   262→   262→    setBusy(true);
   263→   263→    try {
   264→   264→      await api("/api/auth/password", {
   265→   265→        method: "PATCH",
   266→   266→        body: JSON.stringify({ currentPassword: currentPw, newPassword: newPw }),
   267→   267→      });
   268→   268→      toast.success("Password changed");
   269→   269→      setChanging(false);
   270→   270→      setCurrentPw("");
   271→   271→      setNewPw("");
   272→   272→      setConfirmPw("");
   273→   273→    } catch (err) {
   274→   274→      toast.error(err instanceof ApiError ? err.message : "Failed to change password");
   275→   275→    } finally {
   276→   276→      setBusy(false);
   277→   277→    }
   278→   278→  }
   279→   279→
   280→   280→  return (
   281→   281→    <div className="space-y-4">
   282→   282→      <div className="flex items-center justify-between">
   283→   283→        <div>
   284→   284→          <p className="text-sm font-medium">Password</p>
   285→   285→          <p className="text-sm text-muted-foreground">
   286→   286→            {changing ? "Enter your current and new password." : "Change your account password."}
   287→   287→          </p>
   288→   288→        </div>
   289→   289→        {!changing && (
   290→   290→          <Button size="sm" variant="outline" onClick={() => setChanging(true)}>
   291→   291→            <Key className="mr-1.5 size-3.5" /> Change
   292→   292→          </Button>
   293→   293→        )}
   294→   294→      </div>
   295→   295→
   296→   296→      {changing && (
   297→   297→        <div className="space-y-3">
   298→   298→          <div className="space-y-1.5">
   299→   299→            <Label htmlFor="current-pw">Current password</Label>
   300→   300→            <Input
   301→   301→              id="current-pw"
   302→   302→              type="password"
   303→   303→              value={currentPw}
   304→   304→              onChange={(e) => setCurrentPw(e.target.value)}
   305→   305→            />
   306→   306→          </div>
   307→   307→          <div className="space-y-1.5">
   308→   308→            <Label htmlFor="new-pw">New password</Label>
   309→   309→            <Input
   310→   310→              id="new-pw"
   311→   311→              type="password"
   312→   312→              value={newPw}
   313→   313→              onChange={(e) => setNewPw(e.target.value)}
   314→   314→              minLength={6}
   315→   315→            />
   316→   316→          </div>
   317→   317→          <div className="space-y-1.5">
   318→   318→            <Label htmlFor="confirm-pw">Confirm new password</Label>
   319→   319→            <Input
   320→   320→              id="confirm-pw"
   321→   321→              type="password"
   322→   322→              value={confirmPw}
   323→   323→              onChange={(e) => setConfirmPw(e.target.value)}
   324→   324→              minLength={6}
   325→   325→            />
   326→   326→          </div>
   327→   327→          <div className="flex items-center gap-2">
   328→   328→            <Button size="sm" onClick={changePassword} disabled={busy || !currentPw || !newPw}>
   329→   329→              {busy ? <Loader2 className="mr-1.5 size-3.5 animate-spin" /> : <Check className="mr-1.5 size-3.5" />}
   330→   330→              Update password
   331→   331→            </Button>
   332→   332→            <Button size="sm" variant="ghost" onClick={() => { setChanging(false); setCurrentPw(""); setNewPw(""); setConfirmPw(""); }}>
   333→   333→              Cancel
   334→   334→            </Button>
   335→   335→          </div>
   336→   336→        </div>
   337→   337→      )}
   338→   338→
   339→   339→      <Separator />
   340→   340→
   341→   341→      <div className="flex items-center justify-between">
   342→   342→        <div>
   343→   343→          <p className="text-sm font-medium">Authentication method</p>
   344→   344→          <p className="text-sm text-muted-foreground">Email + password (scrypt-hashed)</p>
   345→   345→        </div>
   346→   346→        <Badge variant="secondary" className="gap-1">
   347→   347→          <Shield className="size-3" /> Secure
   348→   348→        </Badge>
   349→   349→      </div>
   350→   350→    </div>
   351→   351→  );
   352→   352→}

   851→   851→function AccountSection({
