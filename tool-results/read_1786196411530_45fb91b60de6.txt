    63→function statusAccentBorder(status: string): string {
    64→  switch (status) {
    65→    case "running":
    66→      return "border-l-emerald-500";
    67→    case "pending":
    68→    case "creating":
    69→      return "border-l-amber-500";
    70→    case "failed":
    71→    case "dead":
    72→      return "border-l-red-500";
    73→    case "stopped":
    74→    case "exited":
    75→      return "border-l-zinc-400 dark:border-l-zinc-500";
    76→    default:
    77→      return "border-l-muted-foreground";
    78→  }
    79→}
    80→
    81→function statusDotClass(status: string): string {
    82→  switch (status) {
    83→    case "running":
    84→      return "bg-emerald-500";
    85→    case "stopped":
    86→    case "exited":
    87→      return "bg-zinc-400";
    88→    case "pending":
    89→    case "creating":
    90→      return "bg-amber-500";
    91→    case "failed":
    92→    case "dead":
    93→      return "bg-red-500";
    94→    default:
    95→      return "bg-muted-foreground";
    96→  }

    99→export function DashboardView() {
   100→  const [deployments, setDeployments] = useState<DeploymentItem[] | null>(null);
   101→  const [busyId, setBusyId] = useState<string | null>(null);
   102→  const [refreshing, setRefreshing] = useState(false);
   103→  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set());
   104→  const [batchBusy, setBatchBusy] = useState(false);
   105→
   106→  const load = useCallback(async () => {
   107→    try {
   108→      const { deployments } = await api<{ deployments: DeploymentItem[] }>("/api/deployments");
   109→      setDeployments(deployments);
   110→    } catch {
   111→      toast.error("Failed to load deployments");
   112→      setDeployments([]);
   113→    }
   114→  }, []);
   115→
   116→  useEffect(() => {
   117→    load();
   118→    const t = setInterval(load, 5000);
   119→    return () => clearInterval(t);
   120→  }, [load]);
   121→
   122→  async function refresh() {
   123→    setRefreshing(true);
   124→    await load();
   125→    setRefreshing(false);
   126→  }
   127→
   128→  async function act(id: string, action: "start" | "stop" | "restart") {
   129→    setBusyId(id);
   130→    try {
   131→      const { deployment } = await api<{ deployment: DeploymentItem }>(`/api/deployments/${id}/${action}`, {
   132→        method: "POST",
   133→      });
   134→      setDeployments((d) => (d ?? []).map((x) => (x.id === id ? deployment : x)));
   135→      toast.success(`Deployment ${action === 'stop' ? 'stopped' : action + 'ed'}`);
   136→    } catch (e) {
   137→      toast.error(e instanceof ApiError ? e.message : `Failed to ${action}`);
   138→    } finally {
   139→      setBusyId(null);
   140→    }
   141→  }
   142→
   143→  async function remove(id: string) {
   144→    setBusyId(id);
   145→    try {
   146→      await api(`/api/deployments/${id}`, { method: "DELETE" });
   147→      setDeployments((d) => (d ?? []).filter((x) => x.id !== id));
   148→      toast.success("Deployment deleted");
   149→    } catch (e) {
   150→      toast.error(e instanceof ApiError ? e.message : "Failed to delete");
   151→    } finally {
   152→      setBusyId(null);
   153→    }
   154→  }
   155→
   156→  // Batch operations
   157→  function toggleSelect(id: string) {
   158→    setSelectedIds((prev) => {
   159→      const next = new Set(prev);
   160→      if (next.has(id)) next.delete(id);
   161→      else next.add(id);
   162→      return next;
   163→    });
   164→  }
   165→
   166→  function toggleSelectAll() {
   167→    if (!deployments) return;
   168→    if (selectedIds.size === deployments.length) {
   169→      setSelectedIds(new Set());
   170→    } else {
   171→      setSelectedIds(new Set(deployments.map((d) => d.id)));
   172→    }
   173→  }
   174→
   175→  function deselectAll() {
   176→    setSelectedIds(new Set());
   177→  }
   178→
   179→  const batchInfo = useMemo(() => {
   180→    if (!deployments) return { hasStopped: false, hasRunning: false, count: 0 };
   181→    const selected = deployments.filter((d) => selectedIds.has(d.id));
   182→    return {
   183→      hasStopped: selected.some((d) => d.status === "stopped" || d.status === "exited"),
   184→      hasRunning: selected.some((d) => d.status === "running"),
   185→      count: selected.length,
   186→    };
   187→  }, [deployments, selectedIds]);
   188→
   189→  async function batchAction(action: "start" | "stop" | "restart") {
   190→    setBatchBusy(true);
   191→    let successCount = 0;
   192→    let failCount = 0;
   193→    for (const id of selectedIds) {
   194→      try {
   195→        await api<{ deployment: DeploymentItem }>(`/api/deployments/${id}/${action}`, { method: "POST" });
   196→        successCount++;
   197→      } catch {
   198→        failCount++;
   199→      }
   200→    }
   201→    await load();
   202→    setBatchBusy(false);
   203→    deselectAll();
   204→    if (failCount === 0) toast.success(`${successCount} deployment(s) ${action === 'stop' ? 'stopped' : action + 'ed'}`);
   205→    else toast.error(`${failCount} failed, ${successCount} succeeded`);
   206→  }
   207→
   208→  async function batchDelete() {
   209→    setBatchBusy(true);
   210→    let successCount = 0;
   211→    let failCount = 0;
   212→    for (const id of selectedIds) {
   213→      try {
   214→        await api(`/api/deployments/${id}`, { method: "DELETE" });
   215→        successCount++;
   216→      } catch {
   217→        failCount++;
   218→      }
   219→    }
   220→    await load();
   221→    setBatchBusy(false);
   222→    deselectAll();
   223→    if (failCount === 0) toast.success(`${successCount} deployment(s) deleted`);
   224→    else toast.error(`${failCount} failed, ${successCount} succeeded`);
   225→  }
   226→
   227→  return (
   228→    <div className="mx-auto max-w-6xl px-4 py-8">
   229→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   230→        <div>
   231→          <h1 className="text-2xl font-bold tracking-tight">My deployments</h1>
   232→          <p className="text-sm text-muted-foreground">Manage your isolated application instances.</p>
   233→        </div>
   234→        <div className="flex items-center gap-2">
   235→          <Button variant="outline" size="sm" onClick={refresh} disabled={refreshing}>
   236→            {refreshing ? <Loader2 className="mr-2 size-4 animate-spin" /> : <RefreshCw className="mr-2 size-4" />}
   237→            Refresh
   238→          </Button>
   239→          <Button size="sm" className="bg-brand text-brand-foreground hover:bg-brand/90" onClick={() => navigate({ name: "marketplace" })}>
   240→            <Plus className="mr-2 size-4" /> Deploy new
   241→          </Button>
   242→        </div>
   243→      </div>
   244→
   245→      {/* Stats cards */}
   246→      {deployments && deployments.length > 0 && (
   247→        <div className="mt-6 grid grid-cols-2 gap-3 sm:grid-cols-4">
   248→          <StatCard
   249→            icon={<Boxes className="size-4" />}
   250→            label="Total"
   251→            value={String(deployments.length)}
   252→            tone="default"
   253→          />
   254→          <StatCard
   255→            icon={<Activity className="size-4" />}
   256→            label="Running"
   257→            value={String(deployments.filter((d) => d.status === "running").length)}
   258→            tone="emerald"
   259→          />
   260→          <StatCard
   261→            icon={<Square className="size-4" />}
   262→            label="Stopped"
   263→            value={String(deployments.filter((d) => d.status === "stopped" || d.status === "exited").length)}
   264→            tone="zinc"
   265→          />
   266→          <StatCard
   267→            icon={<HardDrive className="size-4" />}
   268→            label="Data stored"
   269→            value={formatDataSize(
   270→              deployments.reduce((acc, d) => acc + (d.volumeDataSize ?? 0), 0)
   271→            )}
   272→            tone="brand"
   273→          />
   274→        </div>
   275→      )}
   276→
   277→      <div className="mt-6">
   278→        {deployments === null ? (
   279→          <div className="space-y-3">
   280→            {Array.from({ length: 2 }).map((_, i) => (
   281→              <div key={i} className="h-28 rounded-2xl shimmer" />
   282→            ))}
   283→          </div>
   284→        ) : deployments.length === 0 ? (
   285→          <EmptyState />
   286→        ) : (
   287→          <div className="space-y-3">
   288→            {deployments.map((d) => (
   289→              <DeploymentRow
   290→                key={d.id}
   291→                d={d}
   292→                busy={busyId === d.id}
   293→                selected={selectedIds.has(d.id)}
   294→                onSelect={() => toggleSelect(d.id)}
   295→                onStart={() => act(d.id, "start")}
   296→                onStop={() => act(d.id, "stop")}
   297→                onRestart={() => act(d.id, "restart")}
   298→                onDelete={() => remove(d.id)}
   299→              />
   300→            ))}
   301→          </div>
   302→        )}
   303→      </div>
   304→
   305→      {/* Resource Usage sparkline section */}
   306→      {deployments && deployments.length > 0 && (
   307→        <ResourceUsageSparklines deployments={deployments} />
   308→      )}
   309→
   310→      {/* Activity timeline */}
   311→      {deployments && deployments.length > 0 && (
   312→        <div className="mt-8">
   313→          <h2 className="mb-3 flex items-center gap-2 text-sm font-semibold uppercase tracking-wider text-muted-foreground">
   314→            <TrendingUp className="size-4" /> Recent activity
   315→          </h2>
   316→          <div className="rounded-2xl border border-border bg-card p-5 shadow-sm">
   317→            <ol className="relative space-y-3">
   318→              <div className="absolute left-[7px] top-2 bottom-2 w-0.5 bg-border" />
   319→              {[...deployments]
   320→                .sort((a, b) => new Date(b.updatedAt).getTime() - new Date(a.updatedAt).getTime())
   321→                .slice(0, 5)
   322→                .map((d) => {
   323→                  const isActive = d.status === "running";
   324→                  return (
   325→                    <li key={d.id} className="relative flex items-start gap-3 pl-6">
   326→                      <span
   327→                        className={`absolute left-0 top-1.5 size-4 rounded-full border-2 border-background ${
   328→                          isActive
   329→                            ? "bg-emerald-500"
   330→                            : d.status === "failed" || d.status === "dead"
   331→                            ? "bg-red-500"
   332→                            : "bg-zinc-400"
   333→                        }`}
   334→                      />
   335→                      <button
   336→                        className="flex flex-1 items-center justify-between gap-2 text-left text-sm hover:opacity-80"
   337→                        onClick={() => navigate({ name: "deployment", id: d.id })}
   338→                      >
   339→                        <span className="min-w-0">
   340→                          <span className="font-medium">{d.app?.name ?? "Unknown app"}</span>{" "}
   341→                          <span className="text-muted-foreground">
   342→                            {isActive ? "is running" : `was ${d.status}`}
   343→                          </span>
   344→                        </span>
   345→                        <span className="shrink-0 text-xs text-muted-foreground">
   346→                          {timeAgo(d.updatedAt)}
   347→                        </span>
   348→                      </button>
   349→                    </li>
   350→                  );
   351→                })}
   352→            </ol>
   353→          </div>
   354→        </div>
   355→      )}
   356→
   357→      {/* Batch action bar - fixed at bottom */}
   358→      {batchInfo.count > 0 && (
   359→        <div className="fixed inset-x-0 bottom-0 z-50 border-t border-border bg-card/95 shadow-lg backdrop-blur-sm animate-in slide-in-from-bottom-4 duration-200">
   360→          <div className="mx-auto flex max-w-6xl items-center justify-between gap-3 px-4 py-3">
   361→            <div className="flex items-center gap-3">
   362→              <Checkbox
   363→                checked={deployments ? selectedIds.size === deployments.length : false}
   364→                onCheckedChange={toggleSelectAll}
   365→                aria-label="Select all"
   366→              />
   367→              <span className="text-sm font-medium">
   368→                {batchInfo.count} selected
   369→              </span>
   370→            </div>
   371→            <div className="flex items-center gap-2">
   372→              <Button
   373→                size="sm"
   374→                variant="outline"
   375→                className="gap-1.5"
   376→                onClick={() => batchAction("start")}
   377→                disabled={batchBusy || !batchInfo.hasStopped}
   378→              >
   379→                {batchBusy ? <Loader2 className="size-3.5 animate-spin" /> : <Play className="size-3.5" />}
   380→                Start all
   381→              </Button>
   382→              <Button
   383→                size="sm"
   384→                variant="outline"
   385→                className="gap-1.5"
   386→                onClick={() => batchAction("stop")}
   387→                disabled={batchBusy || !batchInfo.hasRunning}
   388→              >
   389→                {batchBusy ? <Loader2 className="size-3.5 animate-spin" /> : <Square className="size-3.5" />}
   390→                Stop all
   391→              </Button>
   392→              <Button
   393→                size="sm"
   394→                variant="outline"
   395→                className="gap-1.5"
   396→                onClick={() => batchAction("restart")}
   397→                disabled={batchBusy}
   398→              >
   399→                {batchBusy ? <Loader2 className="size-3.5 animate-spin" /> : <RotateCw className="size-3.5" />}
   400→                Restart all
   401→              </Button>
   402→              <AlertDialog>
   403→                <AlertDialogTrigger asChild>
   404→                  <Button
   405→                    size="sm"
   406→                    variant="outline"
   407→                    className="gap-1.5 border-destructive/30 text-destructive hover:bg-destructive/10 hover:text-destructive"
   408→                    disabled={batchBusy}
   409→                  >
   410→                    <Trash2 className="size-3.5" /> Delete all
   411→                  </Button>
   412→                </AlertDialogTrigger>
   413→                <AlertDialogContent>
   414→                  <AlertDialogHeader>
   415→                    <AlertDialogTitle>Delete {batchInfo.count} deployment(s)?</AlertDialogTitle>
   416→                    <AlertDialogDescription>
   417→                      This permanently removes {batchInfo.count} container(s) and their volumes. All persisted data will be lost.
   418→                    </AlertDialogDescription>
   419→                  </AlertDialogHeader>
   420→                  <AlertDialogFooter>
   421→                    <AlertDialogCancel>Cancel</AlertDialogCancel>
   422→                    <AlertDialogAction
   423→                      onClick={batchDelete}
   424→                      className="bg-destructive text-destructive-foreground hover:bg-destructive/90"
   425→                    >
   426→                      Delete all
   427→                    </AlertDialogAction>
   428→                  </AlertDialogFooter>
   429→                </AlertDialogContent>
   430→              </AlertDialog>
   431→              <Button
   432→                size="sm"
   433→                variant="ghost"
   434→                className="gap-1.5"
   435→                onClick={deselectAll}
   436→              >
   437→                <X className="size-3.5" /> Cancel
   438→              </Button>
   439→            </div>
   440→          </div>
   441→        </div>
   442→      )}
   443→    </div>
   444→  );
   445→}

   447→function StatCard({
   448→  icon,
   449→  label,
   450→  value,
   451→  tone,
   452→}: {
   453→  icon: React.ReactNode;
   454→  label: string;
   455→  value: string;
   456→  tone: "default" | "emerald" | "zinc" | "brand";
   457→}) {
   458→  const toneClasses = {
   459→    default: "bg-muted/60 text-foreground",
   460→    emerald: "bg-emerald-500/10 text-emerald-600 dark:text-emerald-400",
   461→    zinc: "bg-zinc-500/10 text-zinc-600 dark:text-zinc-400",
   462→    brand: "bg-brand-soft text-brand",
   463→  }[tone];
   464→  return (
   465→    <div className="flex items-center gap-3 rounded-xl border border-border bg-card p-3 shadow-sm transition-colors hover:border-brand/30">
   466→      <span className={`flex size-9 items-center justify-center rounded-lg ${toneClasses}`}>
   467→        {icon}
   468→      </span>
   469→      <div className="min-w-0">
   470→        <p className="truncate text-[11px] font-medium uppercase tracking-wider text-muted-foreground">
   471→          {label}
   472→        </p>
   473→        <p className="truncate text-lg font-bold tabular-nums">{value}</p>
   474→      </div>
   475→    </div>
   476→  );
   477→}

   479→function DeploymentRow({
   480→  d,
   481→  busy,
   482→  selected,
   483→  onSelect,
   484→  onStart,
   485→  onStop,
   486→  onRestart,
   487→  onDelete,
   488→}: {
   489→  d: DeploymentItem;
   490→  busy: boolean;
   491→  selected: boolean;
   492→  onSelect: () => void;
   493→  onStart: () => void;
   494→  onStop: () => void;
   495→  onRestart: () => void;
   496→  onDelete: () => void;
   497→}) {
   498→  const running = d.status === "running";
   499→  const containerId = d.containerId ?? d.id;
   500→  const metrics = getContainerMetrics(containerId);
   501→  const uptime = calculateUptime(d.createdAt, d.status);
   502→
   503→  return (
   504→    <div className={`group flex flex-col gap-4 rounded-2xl border border-border border-l-4 ${statusAccentBorder(d.status)} bg-card p-4 shadow-sm transition-colors hover:border-brand/40 sm:flex-row sm:items-center sm:justify-between`}>
   505→      <div className="flex min-w-0 flex-1 items-center gap-3">
   506→        {/* Checkbox */}
   507→        <Checkbox
   508→          checked={selected}
   509→          onCheckedChange={onSelect}
   510→          aria-label={`Select ${d.app?.name ?? "deployment"}`}
   511→          onClick={(e) => e.stopPropagation()}
   512→        />
   513→
   514→        <button
   515→          className="flex min-w-0 flex-1 items-center gap-4 text-left"
   516→          onClick={() => navigate({ name: "deployment", id: d.id })}
   517→        >
   518→          <AppLogo logo={d.app?.logo ?? null} simulator={d.app?.simulator ?? "static"} name={d.app?.name ?? "App"} />
   519→          <div className="min-w-0 flex-1">
   520→            <div className="flex items-center gap-2">
   521→              <h3 className="truncate font-semibold">{d.app?.name ?? "Unknown app"}</h3>
   522→              {d.label && (
   523→                <Badge variant="outline" className="shrink-0 gap-1 border-brand/30 bg-brand-soft/50 px-1.5 py-0 text-[10px] font-medium text-brand">
   524→                  <Tag className="size-2.5" /> {d.label}
   525→                </Badge>
   526→              )}
   527→              <span className={`inline-flex shrink-0 items-center gap-2 rounded-full px-2.5 py-0.5 text-[11px] font-medium ${statusColor(d.status)}`}>
   528→                <span className="relative flex size-1.5">
   529→                  <span className={`absolute inline-flex size-full rounded-full ${statusDotClass(d.status)} opacity-75`} />
   530→                  {d.status === "running" && (
   531→                    <span className={`inline-flex size-full rounded-full ${statusDotClass(d.status)} animate-status-pulse`} />
   532→                  )}
   533→                </span>
   534→                {d.status}
   535→              </span>
   536→              {/* Health dot */}
   537→              <span className="relative flex size-2" title={`Health: ${d.status}`}>
   538→                <span className={`size-full rounded-full ${healthDotColor(d.status)}`} />
   539→              </span>
   540→            </div>
   541→            <div className="mt-0.5 flex flex-wrap items-center gap-x-3 gap-y-0.5 text-xs text-muted-foreground">
   542→              <a
   543→                href={d.previewPath}
   544→                target="_blank"
   545→                rel="noreferrer"
   546→                className="inline-flex max-w-[180px] items-center gap-1 font-mono text-brand hover:underline sm:max-w-[220px]"
   547→                onClick={(e) => e.stopPropagation()}
   548→              >
   549→                <ExternalLink className="size-3 shrink-0" /> <span className="truncate">{d.subdomain}.apps.local</span>
   550→              </a>
   551→              <span className="inline-flex items-center gap-1 font-mono">
   552→                <Container className="size-3" /> <span className="truncate max-w-[150px]">{d.containerName}</span>
   553→              </span>
   554→              <span className="inline-flex items-center gap-1">
   555→                <Timer className="size-3" /> {running ? `Running for ${uptimeSince(d.createdAt)}` : `Stopped · ${timeAgo(d.updatedAt)}`}
   556→              </span>
   557→              {d.volumeDataSize != null && (
   558→                <span className="inline-flex items-center gap-1">
   559→                  <Database className="size-3" /> {formatDataSize(d.volumeDataSize)}
   560→                </span>
   561→              )}
   562→              {/* Uptime indicator */}
   563→              <span className="inline-flex items-center gap-1 tabular-nums">
   564→                <Activity className="size-3" />

   651→function timeAgo(iso: string): string {
   652→  const diff = Date.now() - new Date(iso).getTime();
   653→  const mins = Math.floor(diff / 60000);
   654→  if (mins < 1) return "just now";
   655→  if (mins < 60) return `${mins}m ago`;
   656→  const hrs = Math.floor(mins / 60);
   657→  if (hrs < 24) return `${hrs}h ago`;
   658→  const days = Math.floor(hrs / 24);
   659→  return `${days}d ago`;
   660→}
   661→
   662→function uptimeSince(iso: string): string {
   663→  const diff = Date.now() - new Date(iso).getTime();
   664→  const mins = Math.floor(diff / 60000);
   665→  if (mins < 1) return "<1m";
   666→  if (mins < 60) return `${mins}m`;
   667→  const hrs = Math.floor(mins / 60);
   668→  const remainMins = mins % 60;
   669→  if (hrs < 24) return remainMins > 0 ? `${hrs}h ${remainMins}m` : `${hrs}h`;
   670→  const days = Math.floor(hrs / 24);
   671→  return `${days}d`;
   672→}

   674→function formatDataSize(bytes: number): string {
   675→  if (bytes < 1024) return `${bytes} B`;
   676→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
   677→  return `${(bytes / (1024 * 1024)).toFixed(1)} MB`;
   678→}

   684→const SPARKLINE_POINTS = 20;

   686→function ResourceUsageSparklines({ deployments }: { deployments: DeploymentItem[] }) {
   687→  const runningDeployments = deployments.filter((d) => d.status === "running");
   688→  const hasRunning = runningDeployments.length > 0;
   689→
   690→  // Compute aggregate base metrics across all running deployments
   691→  const aggMetrics = useMemo(() => {
   692→    if (runningDeployments.length === 0) {
   693→      return { cpu: 0, memory: 0, network: 0, disk: 0 };
   694→    }
   695→    let cpu = 0;
   696→    let memory = 0;
   697→    let network = 0;
   698→    let disk = 0;
   699→    for (const d of runningDeployments) {
   700→      const containerId = d.containerId ?? d.id;
   701→      const m = getContainerMetrics(containerId);
   702→      cpu += m.cpuUsagePercent;
   703→      memory += m.memoryUsagePercent;
   704→      // Derive network/disk from other metrics for realism
   705→      network += m.responseLatencyMs / 150 * 40 + 10; // ~10-50 range
   706→      disk += m.healthScore * 0.3 + 5; // ~5-35 range
   707→    }
   708→    const n = runningDeployments.length;
   709→    return {
   710→      cpu: cpu / n,
   711→      memory: memory / n,
   712→      network: network / n,
   713→      disk: disk / n,
   714→    };
   715→  }, [runningDeployments]);
   716→
   717→  // Stable seed for initial series generation
   718→  const seed = useMemo(
   719→    () => `dash-${runningDeployments.map((d) => d.id).join(",")}`,
   720→    [runningDeployments],
   721→  );
   722→
   723→  // Tick counter drives series evolution
   724→  const [tick, setTick] = useState(0);
   725→
   726→  // Auto-refresh every 10 seconds
   727→  useEffect(() => {
   728→    const t = setInterval(() => setTick((n) => n + 1), 10000);
   729→    return () => clearInterval(t);
   730→  }, []);
   731→
   732→  // Compute full series from seed + tick (fully derived, no intermediate state)
   733→  const series = useMemo(() => {
   734→    const cpu = generateTimeSeries(aggMetrics.cpu, SPARKLINE_POINTS, 5, `${seed}:cpu`);
   735→    const memory = generateTimeSeries(aggMetrics.memory, SPARKLINE_POINTS, 4, `${seed}:mem`);
   736→    const network = generateTimeSeries(aggMetrics.network, SPARKLINE_POINTS, 6, `${seed}:net`);
   737→    const disk = generateTimeSeries(aggMetrics.disk, SPARKLINE_POINTS, 4, `${seed}:disk`);
   738→    // Apply tick-based evolution
   739→    let c = cpu, m = memory, n = network, d = disk;
   740→    for (let i = 0; i < tick; i++) {
   741→      c = tickTimeSeries(c, aggMetrics.cpu, 5, i + 1);
   742→      m = tickTimeSeries(m, aggMetrics.memory, 4, i + 1);
   743→      n = tickTimeSeries(n, aggMetrics.network, 6, i + 1);
   744→      d = tickTimeSeries(d, aggMetrics.disk, 4, i + 1);
   745→    }
   746→    return { cpu: c, memory: m, network: n, disk: d };
   747→  }, [aggMetrics, seed, tick]);
   748→
   749→  const color = hasRunning ? "#10b981" : "#a1a1aa";
   750→
   751→  const charts: {
   752→    key: string;
   753→    label: string;
   754→    icon: React.ReactNode;
   755→    data: number[];
   756→    unit: string;
   757→    decimals?: number;
   758→  }[] = [
   759→    {
   760→      key: "cpu",
   761→      label: "CPU Usage",
   762→      icon: <Cpu className="size-3.5" />,
   763→      data: series.cpu,
   764→      unit: "%",
   765→    },
   766→    {
   767→      key: "memory",
   768→      label: "Memory Usage",
   769→      icon: <MemoryStick className="size-3.5" />,
   770→      data: series.memory,
   771→      unit: "%",
   772→    },
   773→    {
   774→      key: "network",
   775→      label: "Network I/O",
   776→      icon: <Wifi className="size-3.5" />,
   777→      data: series.network,
   778→      unit: " MB/s",
   779→      decimals: 1,
   780→    },
   781→    {
   782→      key: "disk",
   783→      label: "Disk I/O",
   784→      icon: <HardDriveUpload className="size-3.5" />,
   785→      data: series.disk,
   786→      unit: " MB/s",
   787→      decimals: 1,
   788→    },
   789→  ];
   790→
   791→  return (
   792→    <div className="mt-8">
   793→      <h2 className="mb-3 flex items-center gap-2 text-sm font-semibold uppercase tracking-wider text-muted-foreground">
   794→        <Activity className="size-4" /> Resource usage
   795→      </h2>
   796→      <div className="grid grid-cols-2 gap-3 sm:grid-cols-4">
   797→        {charts.map((c) => {
   798→          const current = c.data[c.data.length - 1] ?? 0;
   799→          const prev = c.data[c.data.length - 2] ?? current;
   800→          const trendUp = current >= prev;
   801→          const displayVal = c.decimals ? current.toFixed(c.decimals) : Math.round(current);
   802→          return (
   803→            <div
   804→              key={c.key}
   805→              className="flex flex-col rounded-xl border border-border bg-card p-3 shadow-sm transition-colors hover:border-brand/30"
   806→            >
   807→              <div className="flex items-center justify-between">
   808→                <span className="flex items-center gap-1.5 text-[11px] font-medium uppercase tracking-wider text-muted-foreground">
   809→                  {c.icon} {c.label}
   810→                </span>
   811→                <span className={`flex items-center gap-0.5 text-[10px] tabular-nums ${trendUp ? "text-emerald-600 dark:text-emerald-400" : "text-red-500 dark:text-red-400"}`}>
   812→                  {trendUp ? <ArrowUpRight className="size-3" /> : <ArrowDownRight className="size-3" />}
   813→                  {Math.abs(Math.round(current - prev))}
   814→                </span>
   815→              </div>
   816→              <p className="mt-1 text-lg font-bold tabular-nums">
   817→                {displayVal}{c.unit}
   818→              </p>
   819→              <Sparkline
   820→                data={c.data}
   821→                color={color}
   822→                width={100}
   823→                height={24}
   824→                strokeWidth={1.5}
   825→                showArea
   826→                id={`dash-${c.key}`}
   827→                className="mt-1 w-full"
   828→              />
   829→            </div>
   830→          );
   831→        })}
   832→      </div>
   833→    </div>
   834→  );
   835→}
