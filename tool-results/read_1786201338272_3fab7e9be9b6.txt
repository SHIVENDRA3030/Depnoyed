    67→function statusAccentBorder(status: string): string {
    68→  switch (status) {
    69→    case "running":
    70→      return "border-l-emerald-500";
    71→    case "pending":
    72→    case "creating":
    73→      return "border-l-amber-500";
    74→    case "failed":
    75→    case "dead":
    76→      return "border-l-red-500";
    77→    case "stopped":
    78→    case "exited":
    79→      return "border-l-zinc-400 dark:border-l-zinc-500";
    80→    default:
    81→      return "border-l-muted-foreground";
    82→  }
    83→}
    84→
    85→function statusDotClass(status: string): string {
    86→  switch (status) {
    87→    case "running":
    88→      return "bg-emerald-500";
    89→    case "stopped":
    90→    case "exited":
    91→      return "bg-zinc-400";
    92→    case "pending":
    93→    case "creating":
    94→      return "bg-amber-500";
    95→    case "failed":
    96→    case "dead":
    97→      return "bg-red-500";
    98→    default:
    99→      return "bg-muted-foreground";
   100→  }
   101→}
   102→
   103→export function DashboardView() {
   104→  const [deployments, setDeployments] = useState<DeploymentItem[] | null>(null);
   105→  const [busyId, setBusyId] = useState<string | null>(null);
   106→  const [refreshing, setRefreshing] = useState(false);
   107→  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set());
   108→  const [batchBusy, setBatchBusy] = useState(false);
   109→  const [searchQuery, setSearchQuery] = useState("");
   110→  const [statusFilter, setStatusFilter] = useState<string>("all");
   111→
   112→  const load = useCallback(async () => {
   113→    try {
   114→      const { deployments } = await api<{ deployments: DeploymentItem[] }>("/api/deployments");
   115→      setDeployments(deployments);
   116→    } catch {
   117→      toast.error("Failed to load deployments");
   118→      setDeployments([]);
   119→    }
   120→  }, []);
   121→
   122→  useEffect(() => {
   123→    load();
   124→    const t = setInterval(load, 5000);
   125→    return () => clearInterval(t);
   126→  }, [load]);
   127→
   128→  const filteredDeployments = useMemo(() => {
   129→    if (!deployments) return [];
   130→    const q = searchQuery.trim().toLowerCase();
   131→    return deployments.filter((d) => {
   132→      const matchesStatus =
   133→        statusFilter === "all" ||
   134→        (statusFilter === "running" && d.status === "running") ||
   135→        (statusFilter === "stopped" && (d.status === "stopped" || d.status === "exited"));
   136→      if (!matchesStatus) return false;
   137→      if (!q) return true;
   138→      return (
   139→        (d.app?.name?.toLowerCase().includes(q) ?? false) ||
   140→        (d.label?.toLowerCase().includes(q) ?? false) ||
   141→        d.subdomain.toLowerCase().includes(q) ||
   142→        d.containerName.toLowerCase().includes(q)
   143→      );
   144→    });
   145→  }, [deployments, searchQuery, statusFilter]);
   146→
   147→  async function refresh() {
   148→    setRefreshing(true);
   149→    await load();
   150→    setRefreshing(false);
   151→  }
   152→
   153→  async function act(id: string, action: "start" | "stop" | "restart") {
   154→    setBusyId(id);
   155→    try {
   156→      const { deployment } = await api<{ deployment: DeploymentItem }>(`/api/deployments/${id}/${action}`, {
   157→        method: "POST",
   158→      });
   159→      setDeployments((d) => (d ?? []).map((x) => (x.id === id ? deployment : x)));
   160→      toast.success(`Deployment ${action === 'stop' ? 'stopped' : action + 'ed'}`);
   161→    } catch (e) {
   162→      toast.error(e instanceof ApiError ? e.message : `Failed to ${action}`);
   163→    } finally {
   164→      setBusyId(null);
   165→    }
   166→  }
   167→
   168→  async function remove(id: string) {
   169→    setBusyId(id);
   170→    try {
   171→      await api(`/api/deployments/${id}`, { method: "DELETE" });
   172→      setDeployments((d) => (d ?? []).filter((x) => x.id !== id));
   173→      toast.success("Deployment deleted");
   174→    } catch (e) {
   175→      toast.error(e instanceof ApiError ? e.message : "Failed to delete");
   176→    } finally {
   177→      setBusyId(null);
   178→    }
   179→  }
   180→
   181→  // Batch operations
   182→  function toggleSelect(id: string) {
   183→    setSelectedIds((prev) => {
   184→      const next = new Set(prev);
   185→      if (next.has(id)) next.delete(id);
   186→      else next.add(id);
   187→      return next;
   188→    });
   189→  }
   190→
   191→  function toggleSelectAll() {
   192→    if (!deployments) return;
   193→    if (selectedIds.size === deployments.length) {
   194→      setSelectedIds(new Set());
   195→    } else {
   196→      setSelectedIds(new Set(deployments.map((d) => d.id)));
   197→    }
   198→  }
   199→
   200→  function deselectAll() {
   201→    setSelectedIds(new Set());
   202→  }
   203→
   204→  const batchInfo = useMemo(() => {
   205→    if (!deployments) return { hasStopped: false, hasRunning: false, count: 0 };
   206→    const selected = deployments.filter((d) => selectedIds.has(d.id));
   207→    return {
   208→      hasStopped: selected.some((d) => d.status === "stopped" || d.status === "exited"),
   209→      hasRunning: selected.some((d) => d.status === "running"),
   210→      count: selected.length,
   211→    };
   212→  }, [deployments, selectedIds]);
   213→
   214→  async function batchAction(action: "start" | "stop" | "restart") {
   215→    setBatchBusy(true);
   216→    let successCount = 0;
   217→    let failCount = 0;
   218→    for (const id of selectedIds) {
   219→      try {
   220→        await api<{ deployment: DeploymentItem }>(`/api/deployments/${id}/${action}`, { method: "POST" });
   221→        successCount++;
   222→      } catch {
   223→        failCount++;
   224→      }
   225→    }
   226→    await load();
   227→    setBatchBusy(false);
   228→    deselectAll();
   229→    if (failCount === 0) toast.success(`${successCount} deployment(s) ${action === 'stop' ? 'stopped' : action + 'ed'}`);
   230→    else toast.error(`${failCount} failed, ${successCount} succeeded`);
   231→  }
   232→
   233→  async function batchDelete() {
   234→    setBatchBusy(true);
   235→    let successCount = 0;
   236→    let failCount = 0;
   237→    for (const id of selectedIds) {
   238→      try {
   239→        await api(`/api/deployments/${id}`, { method: "DELETE" });
   240→        successCount++;
   241→      } catch {
   242→        failCount++;
   243→      }
   244→    }
   245→    await load();
   246→    setBatchBusy(false);
   247→    deselectAll();
   248→    if (failCount === 0) toast.success(`${successCount} deployment(s) deleted`);
   249→    else toast.error(`${failCount} failed, ${successCount} succeeded`);
   250→  }
   251→
   252→  return (
   253→    <div className="mx-auto max-w-6xl px-4 py-8">
   254→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   255→        <div>
   256→          <h1 className="text-2xl font-bold tracking-tight">My deployments</h1>
   257→          <p className="text-sm text-muted-foreground">Manage your isolated application instances.</p>
   258→        </div>
   259→        <div className="flex items-center gap-2">
   260→          <Button variant="outline" size="sm" onClick={refresh} disabled={refreshing}>
   261→            {refreshing ? <Loader2 className="mr-2 size-4 animate-spin" /> : <RefreshCw className="mr-2 size-4" />}
   262→            Refresh
   263→          </Button>
   264→          <Button size="sm" className="bg-brand text-brand-foreground hover:bg-brand/90" onClick={() => navigate({ name: "marketplace" })}>
   265→            <Plus className="mr-2 size-4" /> Deploy new
   266→          </Button>
   267→        </div>
   268→      </div>
   269→
   270→      {/* Stats cards */}
   271→      {deployments && deployments.length > 0 && (
   272→        <div className="mt-6 grid grid-cols-2 gap-3 sm:grid-cols-4">
   273→          <StatCard
   274→            icon={<Boxes className="size-4" />}
   275→            label="Total"
   276→            value={String(deployments.length)}
   277→            tone="default"
   278→          />
   279→          <StatCard
   280→            icon={<Activity className="size-4" />}
   281→            label="Running"
   282→            value={String(deployments.filter((d) => d.status === "running").length)}
   283→            tone="emerald"
   284→          />
   285→          <StatCard
   286→            icon={<Square className="size-4" />}
   287→            label="Stopped"
   288→            value={String(deployments.filter((d) => d.status === "stopped" || d.status === "exited").length)}
   289→            tone="zinc"
   290→          />
   291→          <StatCard
   292→            icon={<HardDrive className="size-4" />}
   293→            label="Data stored"
   294→            value={formatDataSize(
   295→              deployments.reduce((acc, d) => acc + (d.volumeDataSize ?? 0), 0)
   296→            )}
   297→            tone="brand"
   298→          />
   299→        </div>
   300→      )}
   301→
   302→      {/* Cost estimation */}
   303→      {deployments && deployments.length > 0 && (
   304→        <div className="mt-4 rounded-2xl border border-border bg-gradient-to-br from-card via-card to-emerald-50/30 p-5 shadow-sm dark:to-emerald-950/10">
   305→          <div className="flex items-center justify-between">
   306→            <div className="flex items-center gap-2">
   307→              <span className="flex size-8 items-center justify-center rounded-lg bg-brand-soft text-brand">
   308→                <DollarSign className="size-4" />
   309→              </span>
   310→              <div>
   311→                <h3 className="text-sm font-semibold">Estimated usage cost</h3>
   312→                <p className="text-[11px] text-muted-foreground">Based on running deployments × resource usage</p>
   313→              </div>
   314→            </div>
   315→            <div className="text-right">
   316→              <p className="text-lg font-bold">
   317→                ${estimateCost(deployments).monthly.toFixed(2)}
   318→                <span className="ml-1 text-xs font-normal text-muted-foreground">/mo</span>
   319→              </p>
   320→              <p className="text-[11px] text-muted-foreground">
   321→                ${estimateCost(deployments).daily.toFixed(3)}/day
   322→              </p>
   323→            </div>
   324→          </div>
   325→          <div className="mt-3 flex flex-wrap gap-3">
   326→            <div className="flex items-center gap-1.5 rounded-full border border-border/60 bg-background/60 px-2.5 py-1 text-xs">
   327→              <Cpu className="size-3 text-brand" />
   328→              <span className="text-muted-foreground">CPU:</span>
   329→              <span className="font-medium">${(0.5 * deployments.filter((d) => d.status === "running").length * 5).toFixed(2)}/mo</span>
   330→            </div>
   331→            <div className="flex items-center gap-1.5 rounded-full border border-border/60 bg-background/60 px-2.5 py-1 text-xs">
   332→              <MemoryStick className="size-3 text-brand" />
   333→              <span className="text-muted-foreground">Memory:</span>
   334→              <span className="font-medium">${(0.5 * deployments.filter((d) => d.status === "running").length * 3).toFixed(2)}/mo</span>
   335→            </div>
   336→            <div className="flex items-center gap-1.5 rounded-full border border-border/60 bg-background/60 px-2.5 py-1 text-xs">
   337→              <HardDrive className="size-3 text-brand" />
   338→              <span className="text-muted-foreground">Storage:</span>
   339→              <span className="font-medium">${(Math.max(1, deployments.reduce((a, d) => a + (d.volumeDataSize ?? 0), 0) / 1024 / 1024) * 0.10).toFixed(2)}/mo</span>
   340→            </div>
   341→          </div>
   342→          <p className="mt-3 text-[10px] text-muted-foreground/70">
   343→            Estimation only — actual costs may vary based on provider pricing and usage patterns.
   344→          </p>
   345→        </div>
   346→      )}
   347→
   348→      {/* Search & filter bar */}
   349→      {deployments && deployments.length > 0 && (
   350→        <div className="mt-4 flex flex-col gap-2 sm:flex-row sm:items-center">
   351→          <div className="relative flex-1">
   352→            <Search className="pointer-events-none absolute left-3 top-1/2 size-4 -translate-y-1/2 text-muted-foreground" />
   353→            <Input
   354→              value={searchQuery}
   355→              onChange={(e) => setSearchQuery(e.target.value)}
   356→              placeholder="Search by app name, label, or subdomain…"
   357→              className="pl-9"
   358→            />
   359→            {searchQuery && (
   360→              <button
   361→                onClick={() => setSearchQuery("")}
   362→                className="absolute right-2 top-1/2 -translate-y-1/2 rounded p-1 text-muted-foreground hover:bg-muted hover:text-foreground"
   363→              >
   364→                <X className="size-3.5" />
   365→              </button>
   366→            )}
   367→          </div>
   368→          <div className="flex items-center gap-1.5">
   369→            {["all", "running", "stopped"].map((f) => (
   370→              <button
   371→                key={f}
   372→                onClick={() => setStatusFilter(f)}
   373→                className={`inline-flex items-center gap-1.5 rounded-full border px-3 py-1.5 text-xs font-medium transition-all ${
   374→                  statusFilter === f
   375→                    ? "border-brand/40 bg-brand/10 text-brand"
   376→                    : "border-border bg-background text-muted-foreground hover:bg-muted"
   377→                }`}
   378→              >
   379→                {f === "all" && <Boxes className="size-3" />}
   380→                {f === "running" && <Activity className="size-3" />}
   381→                {f === "stopped" && <Square className="size-3" />}
   382→                {f.charAt(0).toUpperCase() + f.slice(1)}
   383→                <span className="ml-0.5 text-[10px] text-muted-foreground/70">
   384→                  {f === "all"
   385→                    ? deployments.length
   386→                    : f === "running"
   387→                    ? deployments.filter((d) => d.status === "running").length
   388→                    : deployments.filter((d) => d.status === "stopped" || d.status === "exited").length}
   389→                </span>
   390→              </button>
   391→            ))}
   392→          </div>
   393→        </div>
   394→      )}
   395→
   396→      <div className="mt-6">
   397→        {deployments === null ? (
   398→          <div className="space-y-3">
   399→            {Array.from({ length: 2 }).map((_, i) => (
   400→              <div key={i} className="h-28 rounded-2xl shimmer" />
   401→            ))}
   402→          </div>
   403→        ) : deployments.length === 0 ? (
   404→          <EmptyState />
   405→        ) : filteredDeployments.length === 0 ? (
   406→          <div className="flex flex-col items-center rounded-2xl border border-dashed border-border py-12 text-center">
   407→            <Search className="size-8 text-muted-foreground/40" />
   408→            <p className="mt-2 text-sm text-muted-foreground">No deployments match your search.</p>
   409→            <Button variant="outline" size="sm" className="mt-3" onClick={() => { setSearchQuery(""); setStatusFilter("all"); }}>
   410→              Clear filters
   411→            </Button>
   412→          </div>
   413→        ) : (
   414→          <div className="space-y-3">
   415→            {filteredDeployments.map((d) => (
   416→              <DeploymentRow
   417→                key={d.id}
   418→                d={d}
   419→                busy={busyId === d.id}
   420→                selected={selectedIds.has(d.id)}
   421→                onSelect={() => toggleSelect(d.id)}
   422→                onStart={() => act(d.id, "start")}
   423→                onStop={() => act(d.id, "stop")}
   424→                onRestart={() => act(d.id, "restart")}
   425→                onDelete={() => remove(d.id)}
   426→              />
   427→            ))}
   428→          </div>
   429→        )}
   430→      </div>
   431→
   432→      {/* Resource Usage sparkline section */}
   433→      {deployments && deployments.length > 0 && (
   434→        <ResourceUsageSparklines deployments={deployments} />
   435→      )}
   436→
   437→      {/* Activity timeline - Enhanced */}
   438→      {deployments && deployments.length > 0 && (
   439→        <div className="mt-8">
   440→          <div className="flex items-center justify-between">
   441→            <h2 className="flex items-center gap-2 text-sm font-semibold uppercase tracking-wider text-muted-foreground">
   442→              <TrendingUp className="size-4" /> Recent activity
   443→            </h2>
   444→            <span className="text-[11px] text-muted-foreground/60">Live · auto-refresh</span>
   445→          </div>
   446→          <div className="mt-3 overflow-hidden rounded-2xl border border-border bg-card shadow-sm">
   447→            <ol className="relative divide-y divide-border/40">
   448→              {/* Animated timeline spine */}
   449→              {[...deployments]
   450→                .sort((a, b) => new Date(b.updatedAt).getTime() - new Date(a.updatedAt).getTime())
   451→                .slice(0, 6)
   452→                .map((d, i) => {
   453→                  const isActive = d.status === "running";
   454→                  const isFailed = d.status === "failed" || d.status === "dead";
   455→                  const statusIcon = isActive ? "🟢" : isFailed ? "🔴" : "⚪";
   456→                  const statusLabel = isActive ? "Running" : d.status === "stopped" || d.status === "exited" ? "Stopped" : isFailed ? "Failed" : d.status;
   457→                  return (
   458→                    <li key={d.id} className="animate-slide-in-right" style={{ animationDelay: `${i * 60}ms` }}>
   459→                      <button
   460→                        className="group flex w-full items-center gap-3 px-4 py-3 text-left transition-colors hover:bg-muted/30"
   461→                        onClick={() => navigate({ name: "deployment", id: d.id })}
   462→                      >
   463→                        {/* Status indicator with health ring */}
   464→                        <span className="relative flex size-8 shrink-0 items-center justify-center">
   465→                          {isActive && (
   466→                            <span className="absolute inset-0 rounded-full bg-emerald-500/20 animate-health-ring" />
   467→                          )}
   468→                          <span className={`relative flex size-3 rounded-full ${
   469→                            isActive ? "bg-emerald-500" : isFailed ? "bg-red-500" : "bg-zinc-400 dark:bg-zinc-500"
   470→                          }`}>
   471→                            {isActive && (
   472→                              <span className="absolute inset-0 rounded-full bg-emerald-500 animate-status-pulse" />
   473→                            )}
   474→                          </span>
   475→                        </span>
   476→                        {/* App info */}
   477→                        <div className="min-w-0 flex-1">
   478→                          <div className="flex items-center gap-2">
   479→                            <span className="truncate text-sm font-medium group-hover:text-emerald-600 dark:group-hover:text-emerald-400 transition-colors">
   480→                              {d.app?.name ?? "Unknown app"}
   481→                            </span>
   482→                            <Badge
   483→                              variant="secondary"
   484→                              className={`shrink-0 gap-1 text-[10px] font-normal ${
   485→                                isActive
   486→                                  ? "bg-emerald-500/10 text-emerald-600 dark:text-emerald-400"
   487→                                  : isFailed
   488→                                  ? "bg-red-500/10 text-red-600 dark:text-red-400"
   489→                                  : ""
   490→                              }`}
   491→                            >
   492→                              {statusLabel}
   493→                            </Badge>
   494→                            {d.label && (
   495→                              <Badge variant="outline" className="shrink-0 text-[10px] font-normal">
   496→                                <Tag className="mr-0.5 size-2.5" /> {d.label}
   497→                              </Badge>
   498→                            )}
   499→                          </div>
   500→                          <div className="mt-0.5 flex items-center gap-2 text-[11px] text-muted-foreground/70">
   501→                            <span className="font-mono">{d.subdomain}</span>
   502→                            {d.volumeDataSize != null && d.volumeDataSize > 0 && (
   503→                              <span>· {formatDataSize(d.volumeDataSize)} stored</span>
   504→                            )}
   505→                          </div>
   506→                        </div>
   507→                        {/* Timestamp */}
   508→                        <div className="shrink-0 text-right">
   509→                          <span className="block text-xs text-muted-foreground">{timeAgo(d.updatedAt)}</span>
   510→                          <span className="block text-[10px] text-muted-foreground/50">
   511→                            {new Date(d.updatedAt).toLocaleTimeString([], { hour: "2-digit", minute: "2-digit" })}
   512→                          </span>
   513→                        </div>
   514→                      </button>
   515→                    </li>
   516→                  );
   517→                })}
   518→            </ol>
   519→            {deployments.length > 6 && (
   520→              <div className="border-t border-border/40 px-4 py-2.5 text-center">
   521→                <button
   522→                  onClick={() => navigate({ name: "dashboard" })}
   523→                  className="text-xs font-medium text-muted-foreground hover:text-foreground transition-colors"

   850→function timeAgo(iso: string): string {
   851→  const diff = Date.now() - new Date(iso).getTime();
   852→  const mins = Math.floor(diff / 60000);
   853→  if (mins < 1) return "just now";
   854→  if (mins < 60) return `${mins}m ago`;
   855→  const hrs = Math.floor(mins / 60);
   856→  if (hrs < 24) return `${hrs}h ago`;
   857→  const days = Math.floor(hrs / 24);
   858→  return `${days}d ago`;
   859→}
   860→
   861→function uptimeSince(iso: string): string {
   862→  const diff = Date.now() - new Date(iso).getTime();
   863→  const mins = Math.floor(diff / 60000);
   864→  if (mins < 1) return "<1m";
   865→  if (mins < 60) return `${mins}m`;
   866→  const hrs = Math.floor(mins / 60);
   867→  const remainMins = mins % 60;
   868→  if (hrs < 24) return remainMins > 0 ? `${hrs}h ${remainMins}m` : `${hrs}h`;
   869→  const days = Math.floor(hrs / 24);
   870→  return `${days}d`;
   871→}

   873→function formatDataSize(bytes: number): string {
   874→  if (bytes < 1024) return `${bytes} B`;
   875→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
   876→  return `${(bytes / (1024 * 1024)).toFixed(1)} MB`;
   877→}

   879→function estimateCost(deployments: DeploymentItem[]) {
   880→  const running = deployments.filter((d) => d.status === "running").length;
   881→  const cpuCost = 0.5 * running * 5;  // $5/core/month
   882→  const memCost = 0.5 * running * 3;   // $3/512MB/month
   883→  const totalStorage = deployments.reduce((a, d) => a + (d.volumeDataSize ?? 0), 0);
   884→  const storageGB = Math.max(1, totalStorage / 1024 / 1024);
   885→  const storageCost = storageGB * 0.10; // $0.10/GB/month
   886→  const monthly = cpuCost + memCost + storageCost;
   887→  const daily = monthly / 30;
   888→  return { monthly, daily };
   889→}

   897→function ResourceUsageSparklines({ deployments }: { deployments: DeploymentItem[] }) {
   898→  const runningDeployments = deployments.filter((d) => d.status === "running");
   899→  const hasRunning = runningDeployments.length > 0;
   900→
   901→  // Compute aggregate base metrics across all running deployments
   902→  const aggMetrics = useMemo(() => {
   903→    if (runningDeployments.length === 0) {
   904→      return { cpu: 0, memory: 0, network: 0, disk: 0 };
   905→    }
   906→    let cpu = 0;
   907→    let memory = 0;
   908→    let network = 0;
   909→    let disk = 0;
   910→    for (const d of runningDeployments) {
   911→      const containerId = d.containerId ?? d.id;
   912→      const m = getContainerMetrics(containerId);
   913→      cpu += m.cpuUsagePercent;
   914→      memory += m.memoryUsagePercent;
   915→      // Derive network/disk from other metrics for realism
   916→      network += m.responseLatencyMs / 150 * 40 + 10; // ~10-50 range
   917→      disk += m.healthScore * 0.3 + 5; // ~5-35 range
   918→    }
   919→    const n = runningDeployments.length;
   920→    return {
   921→      cpu: cpu / n,
   922→      memory: memory / n,
   923→      network: network / n,
   924→      disk: disk / n,
   925→    };
   926→  }, [runningDeployments]);
   927→
   928→  // Stable seed for initial series generation
   929→  const seed = useMemo(
   930→    () => `dash-${runningDeployments.map((d) => d.id).join(",")}`,
   931→    [runningDeployments],
   932→  );
   933→
   934→  // Tick counter drives series evolution
   935→  const [tick, setTick] = useState(0);
   936→
   937→  // Auto-refresh every 10 seconds
   938→  useEffect(() => {
   939→    const t = setInterval(() => setTick((n) => n + 1), 10000);
   940→    return () => clearInterval(t);

   953→      m = tickTimeSeries(m, aggMetrics.memory, 4, i + 1);

   955→      d = tickTimeSeries(d, aggMetrics.disk, 4, i + 1);

   957→    return { cpu: c, memory: m, network: n, disk: d };

   990→      decimals: 1,
   991→    },
   992→    {
   993→      key: "disk",
   994→      label: "Disk I/O",

   996→      data: series.disk,
   997→      unit: " MB/s",
   998→      decimals: 1,
   999→    },
  1000→  ];
  1001→
  1002→  return (
  1003→    <div className="mt-8">
  1004→      <h2 className="mb-3 flex items-center gap-2 text-sm font-semibold uppercase tracking-wider text-muted-foreground">
