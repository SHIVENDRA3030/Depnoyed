    23→/* ----------------------------- AnimatedCounter ----------------------------- */
    24→
    25→function AnimatedCounter({ target, duration = 1200, className = "" }: { target: number; duration?: number; className?: string }) {
    26→  const [display, setDisplay] = useState(0);
    27→  const ref = useRef<{ start: number | null; raf: number }>({ start: null, raf: 0 });
    28→
    29→  useEffect(() => {
    30→    const startTime = performance.now();
    31→    ref.current.start = startTime;
    32→
    33→    function step(now: number) {
    34→      const elapsed = now - startTime;
    35→      const progress = Math.min(elapsed / duration, 1);
    36→      // ease-out cubic for a nice deceleration
    37→      const eased = 1 - Math.pow(1 - progress, 3);
    38→      setDisplay(Math.round(eased * target));
    39→      if (progress < 1) {
    40→        ref.current.raf = requestAnimationFrame(step);
    41→      }
    42→    }
    43→
    44→    ref.current.raf = requestAnimationFrame(step);
    45→    return () => cancelAnimationFrame(ref.current.raf);
    46→  }, [target, duration]);
    47→
    48→  return <span className={className}>{display}</span>;
    49→}

    64→function getFavorites(): string[] {
    65→  if (typeof window === "undefined") return [];
    66→  try { return JSON.parse(localStorage.getItem("oss-favorites") || "[]"); } catch { return []; }
    67→}

    69→function getRecentSearches(): string[] {
    70→  if (typeof window === "undefined") return [];
    71→  try { return JSON.parse(localStorage.getItem("oss-recent-searches") || "[]"); } catch { return []; }
    72→}

    74→function addRecentSearch(q: string) {
    75→  if (!q.trim()) return;
    76→  try {
    77→    const existing = getRecentSearches().filter(s => s !== q);
    78→    const next = [q, ...existing].slice(0, 5);
    79→    localStorage.setItem("oss-recent-searches", JSON.stringify(next));
    80→  } catch {}
    81→}
    82→
    83→function clearRecentSearches() {
    84→  try { localStorage.removeItem("oss-recent-searches"); } catch {}
    85→}

   561→function AppCard({ app, onFavToggle }: { app: AppItem; onFavToggle?: () => void }) {
   562→  const user = useAuth((s) => s.user);
   563→  const [modalOpen, setModalOpen] = useState(false);
   564→  const inCompare = useCompare((s) => s.has(app.slug));
   565→  const toggleCompare = useCompare((s) => s.toggle);
   566→  const compareCount = useCompare((s) => s.slugs.length);
   567→  const [fav, setFav] = useState(() => {
   568→    if (typeof window === "undefined") return false;
   569→    try {
   570→      return JSON.parse(localStorage.getItem("oss-favorites") || "[]").includes(app.slug);
   571→    } catch { return false; }
   572→  });
   573→
   574→  function handleDeploy(e: React.MouseEvent) {
   575→    e.stopPropagation();
   576→    e.preventDefault();
   577→    if (!user) {
   578→      toast.info("Sign in to deploy");
   579→      navigate({ name: "login" });
   580→      return;
   581→    }
   582→    setModalOpen(true);
   583→  }
   584→
   585→  function toggleFav(e: React.MouseEvent) {
   586→    e.stopPropagation();
   587→    e.preventDefault();
   588→    try {
   589→      const favs: string[] = JSON.parse(localStorage.getItem("oss-favorites") || "[]");
   590→      const next = favs.includes(app.slug) ? favs.filter(f => f !== app.slug) : [...favs, app.slug];
   591→      localStorage.setItem("oss-favorites", JSON.stringify(next));
   592→      setFav(!fav);
   593→      onFavToggle?.();
   594→    } catch {}
   595→  }
   596→
   597→  return (
   598→    <div
   599→      onClick={() => navigate({ name: "app", slug: app.slug })}
   600→      className="group relative flex cursor-pointer flex-col overflow-hidden rounded-2xl border border-border bg-card p-0 text-left shadow-sm transition-all duration-300 hover:-translate-y-0.5 hover:border-emerald-500/30 hover:shadow-lg card-glow-hover"
   601→    >
   602→      {/* Top accent bar with gradient animation */}
   603→      <div className="h-1 w-full bg-gradient-to-r from-emerald-500/0 via-emerald-500/40 to-teal-500/0 opacity-0 transition-opacity duration-300 group-hover:opacity-100" />
   604→      
   605→      <div className="flex flex-1 flex-col p-5">
   606→        {/* Header */}
   607→        <div className="flex items-start justify-between">
   608→          <div className="relative">
   609→            <AppLogo logo={app.logo} simulator={app.simulator} name={app.name} size="lg" />
   610→            {fav && (
   611→              <span className="absolute -right-1 -top-1 flex size-3 items-center justify-center">
   612→                <Heart className="size-2.5 fill-rose-500 text-rose-500" />
   613→              </span>
   614→            )}
   615→          </div>
   616→          <div className="flex items-center gap-1.5">
   617→            {app.deploymentCount > 0 && (
   618→              <Badge variant="secondary" className="gap-1 font-normal">
   619→                <Zap className="size-3 text-emerald-500" /> {app.deploymentCount}
   620→              </Badge>
   621→            )}
   622→            <Badge variant="secondary" className="font-normal">{app.category}</Badge>
   623→          </div>
   624→        </div>
   625→        <h3 className="mt-4 text-base font-semibold group-hover:text-emerald-600 dark:group-hover:text-emerald-400 transition-colors duration-200">{app.name}</h3>
   626→        <p className="mt-1.5 line-clamp-2 flex-1 text-sm text-muted-foreground">{app.description}</p>
   627→        {/* Version & repo info */}
   628→        <div className="mt-3 flex items-center gap-3 text-[11px] text-muted-foreground/60">
   629→          {app.version && (
   630→            <span className="inline-flex items-center gap-1">
   631→              <span className="size-1.5 rounded-full bg-emerald-500/40" /> v{app.version}
   632→            </span>
   633→          )}
   634→          {app.repository && (
   635→            <span className="inline-flex items-center gap-1">
   636→              <Globe2 className="size-3" /> Repository
   637→            </span>
   638→          )}
   639→        </div>
   640→      </div>
   641→
   642→      {/* Footer */}
   643→      <div className="relative z-20 flex items-center justify-between border-t border-border/40 bg-muted/20 px-5 py-3">
   644→        <span className="font-mono text-[11px] text-muted-foreground/70">{app.dockerImage}</span>
   645→        <div className="flex items-center gap-2">
   646→          <button
   647→            onClick={toggleFav}
   648→            className={`inline-flex items-center justify-center rounded-md p-1 transition-all duration-200 ${fav ? "text-rose-500 hover:text-rose-600" : "text-muted-foreground/60 hover:bg-muted hover:text-rose-500"}`}
   649→            aria-label={fav ? "Remove from favorites" : "Add to favorites"}
   650→          >
   651→            <Heart className={`size-3.5 transition-all duration-200 ${fav ? "fill-rose-500" : ""}`} />
   652→          </button>
   653→          <button
   654→            onClick={(e) => { e.stopPropagation(); e.preventDefault(); toggleCompare(app.slug); if (!inCompare && compareCount >= 3) toast.info("Compare tray full (max 3)"); }}
   655→            className={`inline-flex items-center justify-center rounded-md p-1 transition-colors ${inCompare ? "text-brand bg-brand-soft" : "text-muted-foreground/60 hover:bg-muted hover:text-brand"}`}
   656→            aria-label={inCompare ? "Remove from compare" : "Add to compare"}
   657→            title={inCompare ? "Remove from compare" : "Add to compare"}
   658→          >
   659→            <GitCompare className="size-3.5" />
   660→          </button>
   661→          <button
   662→            onClick={handleDeploy}
   663→            className="deploy-glow-btn inline-flex items-center gap-1.5 rounded-lg bg-emerald-500/10 px-3 py-1.5 text-xs font-semibold text-emerald-600 transition-all duration-200 hover:bg-emerald-500/20 hover:shadow-sm dark:text-emerald-400"
   664→          >
   665→            <Rocket className="size-3" /> Deploy
   666→          </button>
   667→          <span className="inline-flex items-center gap-1 text-xs font-medium text-muted-foreground/60 transition-all duration-200 group-hover:translate-x-0.5 group-hover:text-muted-foreground">
   668→            Details <ArrowRight className="size-3" />
   669→          </span>
   670→        </div>
   671→      </div>
   672→      <DeployModal app={app} open={modalOpen} onOpenChange={setModalOpen} />
   673→    </div>
   674→  );
   675→}
