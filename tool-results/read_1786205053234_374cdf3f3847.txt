   164→function ProfileSection({
   165→  user,
   166→  hydrate,
   167→}: {
   168→  user: NonNullable<ReturnType<typeof useAuth.getState>["user"]>;
   169→  hydrate: () => Promise<void>;
   170→}) {
   171→  const [editing, setEditing] = useState(false);
   172→  const [name, setName] = useState(user.name ?? "");
   173→  const [busy, setBusy] = useState(false);
   174→
   175→  async function saveProfile() {
   176→    setBusy(true);
   177→    try {
   178→      await api("/api/auth/profile", {
   179→        method: "PATCH",
   180→        body: JSON.stringify({ name }),
   181→      });
   182→      await hydrate();
   183→      setEditing(false);
   184→      toast.success("Profile updated");
   185→    } catch (err) {
   186→      toast.error(err instanceof ApiError ? err.message : "Failed to update profile");
   187→    } finally {
   188→      setBusy(false);
   189→    }
   190→  }
   191→
   192→  return (
   193→    <div className="space-y-4">
   194→      <div className="flex items-center gap-4">
   195→        <div className="flex size-16 items-center justify-center rounded-2xl bg-gradient-to-br from-emerald-500 to-teal-600 text-xl font-bold text-white shadow-md">
   196→          {user.email.slice(0, 2).toUpperCase()}
   197→        </div>
   198→        <div className="min-w-0 flex-1">
   199→          <p className="text-lg font-semibold">{user.name || user.email}</p>
   200→          <p className="truncate text-sm text-muted-foreground">{user.email}</p>
   201→          <p className="mt-1 text-xs text-muted-foreground">
   202→            Member since {new Date(user.createdAt).toLocaleDateString()}
   203→          </p>
   204→        </div>
   205→      </div>
   206→
   207→      <Separator />
   208→
   209→      {editing ? (
   210→        <div className="space-y-3">
   211→          <div className="space-y-1.5">
   212→            <Label htmlFor="profile-name">Display name</Label>
   213→            <Input
   214→              id="profile-name"
   215→              value={name}
   216→              onChange={(e) => setName(e.target.value)}
   217→              placeholder="Your name"
   218→              maxLength={60}
   219→            />
   220→          </div>
   221→          <div className="flex items-center gap-2">
   222→            <Button size="sm" onClick={saveProfile} disabled={busy || !name.trim()}>
   223→              {busy ? <Loader2 className="mr-1.5 size-3.5 animate-spin" /> : <Check className="mr-1.5 size-3.5" />}
   224→              Save
   225→            </Button>
   226→            <Button size="sm" variant="ghost" onClick={() => setEditing(false)}>
   227→              Cancel
   228→            </Button>
   229→          </div>
   230→        </div>
   231→      ) : (
   232→        <div className="flex items-center justify-between">
   233→          <div>
   234→            <p className="text-sm font-medium">Display name</p>
   235→            <p className="text-sm text-muted-foreground">{user.name || "Not set"}</p>
   236→          </div>
   237→          <Button size="sm" variant="outline" onClick={() => setEditing(true)}>
   238→            <Pencil className="mr-1.5 size-3.5" /> Edit
   239→          </Button>
   240→        </div>
   241→      )}
   242→    </div>
   243→  );
   244→}

   246→function SecuritySection() {
   247→  const [changing, setChanging] = useState(false);
   248→  const [currentPw, setCurrentPw] = useState("");
   249→  const [newPw, setNewPw] = useState("");
   250→  const [confirmPw, setConfirmPw] = useState("");
   251→  const [busy, setBusy] = useState(false);
   252→
   253→  async function changePassword() {
   254→    if (newPw !== confirmPw) {
   255→      toast.error("Passwords don't match");
   256→      return;
   257→    }
   258→    if (newPw.length < 6) {
   259→      toast.error("Password must be at least 6 characters");
   260→      return;
   261→    }
   262→    setBusy(true);
   263→    try {
   264→      await api("/api/auth/password", {
   265→        method: "PATCH",
   266→        body: JSON.stringify({ currentPassword: currentPw, newPassword: newPw }),
   267→      });
   268→      toast.success("Password changed");
   269→      setChanging(false);
   270→      setCurrentPw("");
   271→      setNewPw("");
   272→      setConfirmPw("");
   273→    } catch (err) {
   274→      toast.error(err instanceof ApiError ? err.message : "Failed to change password");
   275→    } finally {
   276→      setBusy(false);
   277→    }
   278→  }
   279→
   280→  return (
   281→    <div className="space-y-4">
   282→      <div className="flex items-center justify-between">
   283→        <div>
   284→          <p className="text-sm font-medium">Password</p>
   285→          <p className="text-sm text-muted-foreground">
   286→            {changing ? "Enter your current and new password." : "Change your account password."}
   287→          </p>
   288→        </div>
   289→        {!changing && (
   290→          <Button size="sm" variant="outline" onClick={() => setChanging(true)}>
   291→            <Key className="mr-1.5 size-3.5" /> Change
   292→          </Button>
   293→        )}
   294→      </div>
   295→
   296→      {changing && (
   297→        <div className="space-y-3">
   298→          <div className="space-y-1.5">
   299→            <Label htmlFor="current-pw">Current password</Label>
   300→            <Input
   301→              id="current-pw"
   302→              type="password"
   303→              value={currentPw}
   304→              onChange={(e) => setCurrentPw(e.target.value)}
   305→            />
   306→          </div>
   307→          <div className="space-y-1.5">
   308→            <Label htmlFor="new-pw">New password</Label>
   309→            <Input
   310→              id="new-pw"
   311→              type="password"
   312→              value={newPw}
   313→              onChange={(e) => setNewPw(e.target.value)}
   314→              minLength={6}
   315→            />
   316→          </div>
   317→          <div className="space-y-1.5">
   318→            <Label htmlFor="confirm-pw">Confirm new password</Label>
   319→            <Input
   320→              id="confirm-pw"
   321→              type="password"
   322→              value={confirmPw}
   323→              onChange={(e) => setConfirmPw(e.target.value)}
   324→              minLength={6}
   325→            />
   326→          </div>
   327→          <div className="flex items-center gap-2">
   328→            <Button size="sm" onClick={changePassword} disabled={busy || !currentPw || !newPw}>
   329→              {busy ? <Loader2 className="mr-1.5 size-3.5 animate-spin" /> : <Check className="mr-1.5 size-3.5" />}
   330→              Update password
   331→            </Button>
   332→            <Button size="sm" variant="ghost" onClick={() => { setChanging(false); setCurrentPw(""); setNewPw(""); setConfirmPw(""); }}>
   333→              Cancel
   334→            </Button>
   335→          </div>
   336→        </div>
   337→      )}
   338→
   339→      <Separator />
   340→
   341→      <div className="flex items-center justify-between">
   342→        <div>
   343→          <p className="text-sm font-medium">Authentication method</p>
   344→          <p className="text-sm text-muted-foreground">Email + password (scrypt-hashed)</p>
   345→        </div>
   346→        <Badge variant="secondary" className="gap-1">
   347→          <Shield className="size-3" /> Secure
   348→        </Badge>
   349→      </div>
   350→    </div>
   351→  );
   352→}

   538→function loadTemplates(): DeployTemplate[] {
   539→  if (typeof window === "undefined") return DEFAULT_TEMPLATES;
   540→  try {
   541→    const raw = localStorage.getItem(TEMPLATES_KEY);
   542→    if (!raw) {
   543→      // Pre-populate with defaults on first access
   544→      localStorage.setItem(TEMPLATES_KEY, JSON.stringify(DEFAULT_TEMPLATES));
   545→      return DEFAULT_TEMPLATES;
   546→    }
   547→    return JSON.parse(raw);
   548→  } catch {
   549→    return DEFAULT_TEMPLATES;
   550→  }
   551→}

   553→function saveTemplates(templates: DeployTemplate[]) {
   554→  localStorage.setItem(TEMPLATES_KEY, JSON.stringify(templates));

   557→function DeploymentTemplatesSection() {
   558→  const [templates, setTemplates] = useState<DeployTemplate[]>(() => loadTemplates());
   559→  const [creating, setCreating] = useState(false);
   560→  const [editingId, setEditingId] = useState<string | null>(null);
   561→  const [editName, setEditName] = useState("");
   562→  const [apps, setApps] = useState<AppItem[]>([]);
   563→  const [newName, setNewName] = useState("");
   564→  const [newAppSlug, setNewAppSlug] = useState("");
   565→  const [newLabel, setNewLabel] = useState("");
   566→  const [newEnvKey, setNewEnvKey] = useState("");
   567→  const [newEnvValue, setNewEnvValue] = useState("");
   568→  const [newEnvVars, setNewEnvVars] = useState<Record<string, string>>({});
   569→
   570→  // Load apps catalog for the select dropdown
   571→  useEffect(() => {
   572→    (async () => {
   573→      try {
   574→        const data = await api<{ apps: AppItem[] }>("/api/apps");
   575→        setApps(data.apps);
   576→      } catch {
   577→        /* ignore */
   578→      }
   579→    })();
   580→  }, []);
   581→
   582→  function handleCreate() {
   583→    if (!newName.trim() || !newAppSlug) {
   584→      toast.error("Template name and app are required");
   585→      return;
   586→    }
   587→    const tpl: DeployTemplate = {
   588→      id: `tpl-${Date.now()}`,
   589→      name: newName.trim(),
   590→      appSlug: newAppSlug,
   591→      labelPrefix: newLabel.trim(),
   592→      envVars: { ...newEnvVars },
   593→    };
   594→    const next = [...templates, tpl];
   595→    setTemplates(next);
   596→    saveTemplates(next);
   597→    setCreating(false);
   598→    setNewName("");
   599→    setNewAppSlug("");
   600→    setNewLabel("");
   601→    setNewEnvVars({});
   602→    toast.success("Template created");
   603→  }
   604→
   605→  function handleDelete(id: string) {
   606→    const next = templates.filter((t) => t.id !== id);
   607→    setTemplates(next);
   608→    saveTemplates(next);
   609→    toast.success("Template deleted");
   610→  }
   611→
   612→  function handleAddEnvVar() {
   613→    if (!newEnvKey.trim()) return;
   614→    setNewEnvVars((prev) => ({ ...prev, [newEnvKey.trim()]: newEnvValue.trim() }));
   615→    setNewEnvKey("");
   616→    setNewEnvValue("");
   617→  }
   618→
   619→  function handleRemoveEnvVar(key: string) {
   620→    setNewEnvVars((prev) => {
   621→      const next = { ...prev };
   622→      delete next[key];
   623→      return next;
   624→    });
   625→  }
   626→
   627→  function startEdit(tpl: DeployTemplate) {
   628→    setEditingId(tpl.id);
   629→    setEditName(tpl.name);
   630→  }
   631→
   632→  function saveEdit(id: string) {
   633→    if (!editName.trim()) return;
   634→    const next = templates.map((t) => (t.id === id ? { ...t, name: editName.trim() } : t));
   635→    setTemplates(next);
   636→    saveTemplates(next);
   637→    setEditingId(null);
   638→    toast.success("Template renamed");
   639→  }
   640→
   641→  function applyTemplate(tpl: DeployTemplate) {
   642→    navigate({ name: "app", slug: tpl.appSlug });
   643→  }
   644→
   645→  return (
   646→    <div className="space-y-4">
   647→      {/* Template list */}
   648→      {templates.length === 0 ? (
   649→        <p className="text-sm text-muted-foreground">No templates yet. Create one to get started.</p>
   650→      ) : (
   651→        <div className="space-y-2">
   652→          {templates.map((tpl) => {
   653→            const app = apps.find((a) => a.slug === tpl.appSlug);
   654→            const envCount = Object.keys(tpl.envVars).length;
   655→            return (
   656→              <div
   657→                key={tpl.id}
   658→                className="group flex items-center justify-between gap-3 rounded-lg border border-border/60 bg-background/40 p-3 transition-colors hover:border-brand/30"
   659→              >
   660→                <div className="min-w-0 flex-1">
   661→                  <div className="flex items-center gap-2">
   662→                    {editingId === tpl.id ? (
   663→                      <div className="flex items-center gap-1.5">
   664→                        <Input
   665→                          value={editName}
   666→                          onChange={(e) => setEditName(e.target.value)}
   667→                          onKeyDown={(e) => {
   668→                            if (e.key === "Enter") saveEdit(tpl.id);
   669→                            if (e.key === "Escape") setEditingId(null);
   670→                          }}
   671→                          className="h-7 w-40 text-xs"
   672→                          autoFocus
   673→                        />
   674→                        <button
   675→                          onClick={() => saveEdit(tpl.id)}
   676→                          className="inline-flex size-7 items-center justify-center rounded-md bg-brand text-brand-foreground hover:bg-brand/90"
   677→                          aria-label="Save"
   678→                        >
   679→                          <Check className="size-3.5" />
   680→                        </button>
   681→                        <button
   682→                          onClick={() => setEditingId(null)}
   683→                          className="inline-flex size-7 items-center justify-center rounded-md border border-border text-muted-foreground hover:bg-muted"
   684→                          aria-label="Cancel"
   685→                        >
   686→                          <X className="size-3.5" />
   687→                        </button>
   688→                      </div>
   689→                    ) : (
   690→                      <span className="text-sm font-medium">{tpl.name}</span>
   691→                    )}
   692→                  </div>
   693→                  <div className="mt-0.5 flex flex-wrap items-center gap-2 text-xs text-muted-foreground">
   694→                    <span className="inline-flex items-center gap-1">
   695→                      <Rocket className="size-3" /> {app?.name ?? tpl.appSlug}
   696→                    </span>
   697→                    {tpl.labelPrefix && (
   698→                      <Badge variant="outline" className="gap-1 px-1.5 py-0 text-[10px] font-medium">
   699→                        {tpl.labelPrefix}
   700→                      </Badge>
   701→                    )}
   702→                    {envCount > 0 && (
   703→                      <span className="inline-flex items-center gap-1">
   704→                        <Container className="size-3" /> {envCount} env var{envCount !== 1 ? "s" : ""}
   705→                      </span>
   706→                    )}
   707→                  </div>
   708→                </div>
   709→                <div className="flex items-center gap-1.5">
   710→                  <Button size="sm" variant="outline" onClick={() => applyTemplate(tpl)} className="gap-1.5">
   711→                    <Rocket className="size-3.5" /> Use
   712→                  </Button>
   713→                  {editingId !== tpl.id && (
   714→                    <Button size="sm" variant="ghost" onClick={() => startEdit(tpl)} className="size-8 p-0">
   715→                      <Pencil className="size-3.5" />
   716→                    </Button>
   717→                  )}
   718→                  <AlertDialog>
   719→                    <AlertDialogTrigger asChild>
   720→                      <Button size="sm" variant="ghost" className="size-8 p-0 text-muted-foreground hover:text-destructive">
   721→                        <Trash2 className="size-3.5" />
   722→                      </Button>
   723→                    </AlertDialogTrigger>
   724→                    <AlertDialogContent>
   725→                      <AlertDialogHeader>
   726→                        <AlertDialogTitle>Delete template &quot;{tpl.name}&quot;?</AlertDialogTitle>
   727→                        <AlertDialogDescription>
   728→                          This will remove this deployment template. This action cannot be undone.
   729→                        </AlertDialogDescription>
   730→                      </AlertDialogHeader>
   731→                      <AlertDialogFooter>
   732→                        <AlertDialogCancel>Cancel</AlertDialogCancel>
   733→                        <AlertDialogAction
   734→                          onClick={() => handleDelete(tpl.id)}
   735→                          className="bg-destructive text-destructive-foreground hover:bg-destructive/90"
   736→                        >
   737→                          Delete
   738→                        </AlertDialogAction>
   739→                      </AlertDialogFooter>
   740→                    </AlertDialogContent>
   741→                  </AlertDialog>
   742→                </div>
   743→              </div>
   744→            );
   745→          })}
   746→        </div>
   747→      )}
   748→
   749→      {/* Create template form */}
   750→      {creating ? (
   751→        <div className="space-y-3 rounded-lg border border-border/60 bg-muted/20 p-4">
   752→          <p className="text-sm font-medium">Create new template</p>
   753→          <div className="space-y-1.5">
   754→            <Label htmlFor="tpl-name">Template name</Label>
   755→            <Input
   756→              id="tpl-name"
   757→              value={newName}
   758→              onChange={(e) => setNewName(e.target.value)}
   759→              placeholder="e.g. Production Counter"
   760→            />
   761→          </div>
   762→          <div className="space-y-1.5">
   763→            <Label htmlFor="tpl-app">App</Label>
   764→            <Select value={newAppSlug} onValueChange={setNewAppSlug}>
   765→              <SelectTrigger id="tpl-app">
   766→                <SelectValue placeholder="Select an app…" />
   767→              </SelectTrigger>
   768→              <SelectContent>
   769→                {apps.map((a) => (
   770→                  <SelectItem key={a.slug} value={a.slug}>
   771→                    {a.name}
   772→                  </SelectItem>
   773→                ))}
   774→              </SelectContent>
   775→            </Select>
   776→          </div>
   777→          <div className="space-y-1.5">
   778→            <Label htmlFor="tpl-label">Label prefix (optional)</Label>
   779→            <Input
   780→              id="tpl-label"
   781→              value={newLabel}
   782→              onChange={(e) => setNewLabel(e.target.value)}
   783→              placeholder="e.g. prod, staging, dev"
   784→            />
   785→          </div>
   786→          <div className="space-y-1.5">
   787→            <Label>Environment variables (optional)</Label>
   788→            <div className="flex gap-2">
   789→              <Input
   790→                value={newEnvKey}
   791→                onChange={(e) => setNewEnvKey(e.target.value)}
   792→                placeholder="KEY"
   793→                className="flex-1"
   794→              />
   795→              <Input
   796→                value={newEnvValue}
   797→                onChange={(e) => setNewEnvValue(e.target.value)}
   798→                placeholder="VALUE"
   799→                className="flex-1"
   800→              />
   801→              <Button size="sm" variant="outline" onClick={handleAddEnvVar} disabled={!newEnvKey.trim()}>
   802→                <Plus className="size-3.5" />
   803→              </Button>
   804→            </div>
   805→            {Object.keys(newEnvVars).length > 0 && (
   806→              <div className="mt-2 space-y-1">
   807→                {Object.entries(newEnvVars).map(([k, v]) => (
   808→                  <div key={k} className="flex items-center gap-2 text-xs">
   809→                    <span className="font-mono font-medium">{k}</span>
   810→                    <span className="text-muted-foreground">=</span>
   811→                    <span className="font-mono text-muted-foreground">{v || '""'}</span>
   812→                    <button
   813→                      onClick={() => handleRemoveEnvVar(k)}
   814→                      className="ml-auto text-muted-foreground hover:text-destructive"
   815→                    >
   816→                      <X className="size-3" />
   817→                    </button>
   818→                  </div>
   819→                ))}
   820→              </div>
   821→            )}
   822→          </div>
   823→          <div className="flex items-center gap-2 pt-1">
   824→            <Button size="sm" onClick={handleCreate} disabled={!newName.trim() || !newAppSlug}>
   825→              <Check className="mr-1.5 size-3.5" /> Create template
   826→            </Button>
   827→            <Button
   828→              size="sm"
   829→              variant="ghost"
   830→              onClick={() => {
   831→                setCreating(false);
   832→                setNewName("");
   833→                setNewAppSlug("");
   834→                setNewLabel("");
   835→                setNewEnvVars({});
   836→              }}
   837→            >
   838→              Cancel
   839→            </Button>
   840→          </div>
   841→        </div>
   842→      ) : (
   843→        <Button size="sm" variant="outline" onClick={() => setCreating(true)}>
   844→          <Plus className="mr-1.5 size-3.5" /> Create Template
   845→        </Button>
   846→      )}
   847→    </div>
   848→  );
   849→}

   851→function AccountSection({
   852→  user,
   853→  logout,
   854→}: {
   855→  user: NonNullable<ReturnType<typeof useAuth.getState>["user"]>;
   856→  logout: () => Promise<void>;
   857→}) {
   858→  const [exporting, setExporting] = useState(false);
   859→
   860→  async function exportData() {
   861→    setExporting(true);
   862→    try {
   863→      const data = await api<{ deployments: DeploymentItem[] }>("/api/deployments");
   864→      const exportObj = {
   865→        user: { email: user.email, name: user.name },
   866→        exportedAt: new Date().toISOString(),
   867→        deployments: data.deployments,
   868→      };
   869→      const blob = new Blob([JSON.stringify(exportObj, null, 2)], { type: "application/json" });
   870→      const url = URL.createObjectURL(blob);
   871→      const a = document.createElement("a");
   872→      a.href = url;
   873→      a.download = `oss-deploy-export-${new Date().toISOString().slice(0, 10)}.json`;
   874→      a.click();
   875→      URL.revokeObjectURL(url);
   876→      toast.success("Data exported");
   877→    } catch {
   878→      toast.error("Failed to export data");
   879→    } finally {
   880→      setExporting(false);
   881→    }
   882→  }
   883→
   884→  return (
   885→    <div className="space-y-4">
   886→      <div className="flex items-center justify-between">
   887→        <div>
   888→          <p className="text-sm font-medium">Export data</p>
   889→          <p className="text-xs text-muted-foreground">Download all your deployment data as JSON</p>
   890→        </div>
   891→        <Button size="sm" variant="outline" onClick={exportData} disabled={exporting}>
   892→          {exporting ? <Loader2 className="mr-1.5 size-3.5 animate-spin" /> : <Download className="mr-1.5 size-3.5" />}
   893→          Export
   894→        </Button>
   895→      </div>
   896→
   897→      <Separator />
   898→
   899→      <div className="flex items-center justify-between">
   900→        <div>
   901→          <p className="text-sm font-medium">Sign out</p>
   902→          <p className="text-xs text-muted-foreground">Sign out of your account on this device</p>
   903→        </div>
   904→        <Button size="sm" variant="outline" onClick={() => { logout(); navigate({ name: "marketplace" }); }}>
   905→          Sign out
   906→        </Button>
   907→      </div>
   908→
   909→      <Separator />
   910→
   911→      <div className="flex items-center justify-between">
   912→        <div>
   913→          <p className="text-sm font-medium text-destructive">Delete account</p>
   914→          <p className="text-xs text-muted-foreground">
   915→            Permanently delete your account and all deployments
   916→          </p>
   917→        </div>
   918→        <AlertDialog>
   919→          <AlertDialogTrigger asChild>
   920→            <Button size="sm" variant="outline" className="border-destructive/30 text-destructive hover:bg-destructive/10">
   921→              <Trash2 className="mr-1.5 size-3.5" /> Delete
   922→            </Button>
   923→          </AlertDialogTrigger>
   924→          <AlertDialogContent>
   925→            <AlertDialogHeader>
   926→              <AlertDialogTitle>Delete your account?</AlertDialogTitle>
   927→              <AlertDialogDescription>
   928→                This will permanently delete your account and all {user.email} deployments.
   929→                This action cannot be undone.
   930→              </AlertDialogDescription>
   931→            </AlertDialogHeader>
   932→            <AlertDialogFooter>
   933→              <AlertDialogCancel>Cancel</AlertDialogCancel>
   934→              <AlertDialogAction className="bg-destructive text-destructive-foreground hover:bg-destructive/90">
   935→                Delete account
   936→              </AlertDialogAction>
   937→            </AlertDialogFooter>
   938→          </AlertDialogContent>
   939→        </AlertDialog>
   940→      </div>
   941→    </div>
   942→  );
   943→}
