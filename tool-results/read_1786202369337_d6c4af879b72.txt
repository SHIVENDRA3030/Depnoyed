    23→    23→function getFavorites(): string[] {
    24→    24→  if (typeof window === "undefined") return [];

    28→    28→function getRecentSearches(): string[] {
    29→    29→  if (typeof window === "undefined") return [];

    31→    31→}
    32→    32→
    33→    33→function addRecentSearch(q: string) {
    34→    34→  if (!q.trim()) return;
    35→    35→  try {
    36→    36→    const existing = getRecentSearches().filter(s => s !== q);
    37→    37→    const next = [q, ...existing].slice(0, 5);
    38→    38→    localStorage.setItem("oss-recent-searches", JSON.stringify(next));
    39→    39→  } catch {}
    40→    40→}

    42→    42→function clearRecentSearches() {
    43→    43→  try { localStorage.removeItem("oss-recent-searches"); } catch {}

   345→   345→function FeaturedAppCard({ app, rank }: { app: AppItem; rank: number }) {
   346→   346→  const user = useAuth((s) => s.user);
   347→   347→  const [modalOpen, setModalOpen] = useState(false);
   348→   348→
   349→   349→  function handleDeploy(e: React.MouseEvent) {
   350→   350→    e.stopPropagation();
   351→   351→    e.preventDefault();
   352→   352→    if (!user) {
   353→   353→      toast.info("Sign in to deploy");
   354→   354→      navigate({ name: "login" });
   355→   355→      return;
   356→   356→    }
   357→   357→    setModalOpen(true);
   358→   358→  }
   359→   359→
   360→   360→  const rankColor =
   361→   361→    rank === 1
   362→   362→      ? "from-amber-400 to-orange-500 text-white"
   363→   363→      : rank === 2
   364→   364→      ? "from-zinc-300 to-zinc-400 text-zinc-900"
   365→   365→      : "from-orange-700 to-amber-800 text-white";
   366→   366→
   367→   367→  const rankGlow =
   368→   368→    rank === 1
   369→   369→      ? "shadow-amber-500/20"
   370→   370→      : rank === 2
   371→   371→      ? "shadow-zinc-400/20"
   372→   372→      : "shadow-orange-500/20";
   373→   373→
   374→   374→  return (
   375→   375→    <div

   417→   417→function AppCard({ app, onFavToggle }: { app: AppItem; onFavToggle?: () => void }) {
   418→   418→  const user = useAuth((s) => s.user);
   419→   419→  const [modalOpen, setModalOpen] = useState(false);
   420→   420→  const inCompare = useCompare((s) => s.has(app.slug));
   421→   421→  const toggleCompare = useCompare((s) => s.toggle);
   422→   422→  const compareCount = useCompare((s) => s.slugs.length);
   423→   423→  const [fav, setFav] = useState(() => {
   424→   424→    if (typeof window === "undefined") return false;
   425→   425→    try {
   426→   426→      return JSON.parse(localStorage.getItem("oss-favorites") || "[]").includes(app.slug);
   427→   427→    } catch { return false; }
   428→   428→  });
   429→   429→
   430→   430→  function handleDeploy(e: React.MouseEvent) {
   431→   431→    e.stopPropagation();
   432→   432→    e.preventDefault();
   433→   433→    if (!user) {
   434→   434→      toast.info("Sign in to deploy");
   435→   435→      navigate({ name: "login" });
   436→   436→      return;
   437→   437→    }
   438→   438→    setModalOpen(true);
   439→   439→  }
   440→   440→
   441→   441→  function toggleFav(e: React.MouseEvent) {
   442→   442→    e.stopPropagation();
   443→   443→    e.preventDefault();
   444→   444→    try {
   445→   445→      const favs: string[] = JSON.parse(localStorage.getItem("oss-favorites") || "[]");
   446→   446→      const next = favs.includes(app.slug) ? favs.filter(f => f !== app.slug) : [...favs, app.slug];
   447→   447→      localStorage.setItem("oss-favorites", JSON.stringify(next));
   448→   448→      setFav(!fav);
   449→   449→      onFavToggle?.();
   450→   450→    } catch {}
   451→   451→  }
   452→   452→
   453→   453→  return (
   454→   454→    <div
   455→   455→      onClick={() => navigate({ name: "app", slug: app.slug })}
   456→   456→      className="group relative flex cursor-pointer flex-col overflow-hidden rounded-2xl border border-border bg-card p-0 text-left shadow-sm transition-all duration-300 hover:-translate-y-0.5 hover:border-emerald-500/30 hover:shadow-lg card-glow-hover"
   457→   457→    >
   458→   458→      {/* Top accent bar with gradient animation */}

   467→   467→              <span className="absolute -right-1 -top-1 flex size-3 items-center justify-center">
   468→   468→                <Heart className="size-2.5 fill-rose-500 text-rose-500" />
   469→   469→              </span>
   470→   470→            )}
   471→   471→          </div>
   472→   472→          <div className="flex items-center gap-1.5">
   473→   473→            {app.deploymentCount > 0 && (
   474→   474→              <Badge variant="secondary" className="gap-1 font-normal">
   475→   475→                <Zap className="size-3 text-emerald-500" /> {app.deploymentCount}
   476→   476→              </Badge>

   483→   483→        {/* Version & repo info */}
