   870→   870→  const mins = Math.floor(diff / 60000);
   871→   871→  if (mins < 1) return "just now";
   872→   872→  if (mins < 60) return `${mins}m ago`;
   873→   873→  const hrs = Math.floor(mins / 60);
   874→   874→  if (hrs < 24) return `${hrs}h ago`;
   875→   875→  const days = Math.floor(hrs / 24);
   876→   876→  return `${days}d ago`;

   881→   881→  const mins = Math.floor(diff / 60000);
   882→   882→  if (mins < 1) return "<1m";
   883→   883→  if (mins < 60) return `${mins}m`;
   884→   884→  const hrs = Math.floor(mins / 60);

   886→   886→  if (hrs < 24) return remainMins > 0 ? `${hrs}h ${remainMins}m` : `${hrs}h`;
   887→   887→  const days = Math.floor(hrs / 24);

   897→   897→function estimateCost(deployments: DeploymentItem[]) {
   898→   898→  const running = deployments.filter((d) => d.status === "running").length;
   899→   899→  const cpuCost = 0.5 * running * 5;  // $5/core/month
   900→   900→  const memCost = 0.5 * running * 3;   // $3/512MB/month
   901→   901→  const totalStorage = deployments.reduce((a, d) => a + (d.volumeDataSize ?? 0), 0);
   902→   902→  const storageGB = Math.max(1, totalStorage / 1024 / 1024);
   903→   903→  const storageCost = storageGB * 0.10; // $0.10/GB/month
   904→   904→  const monthly = cpuCost + memCost + storageCost;
   905→   905→  const daily = monthly / 30;
   906→   906→  return { monthly, daily };

  1328→  1328→const SPARKLINE_POINTS = 20;
  1329→  1329→
  1330→  1330→function ResourceUsageSparklines({ deployments }: { deployments: DeploymentItem[] }) {
  1331→  1331→  const runningDeployments = deployments.filter((d) => d.status === "running");
  1332→  1332→  const hasRunning = runningDeployments.length > 0;
  1333→  1333→
  1334→  1334→  // Compute aggregate base metrics across all running deployments
  1335→  1335→  const aggMetrics = useMemo(() => {
  1336→  1336→    if (runningDeployments.length === 0) {
  1337→  1337→      return { cpu: 0, memory: 0, network: 0, disk: 0 };
  1338→  1338→    }
  1339→  1339→    let cpu = 0;
  1340→  1340→    let memory = 0;
  1341→  1341→    let network = 0;
  1342→  1342→    let disk = 0;
  1343→  1343→    for (const d of runningDeployments) {
  1344→  1344→      const containerId = d.containerId ?? d.id;
  1345→  1345→      const m = getContainerMetrics(containerId);
  1346→  1346→      cpu += m.cpuUsagePercent;
  1347→  1347→      memory += m.memoryUsagePercent;
  1348→  1348→      // Derive network/disk from other metrics for realism
  1349→  1349→      network += m.responseLatencyMs / 150 * 40 + 10; // ~10-50 range
  1350→  1350→      disk += m.healthScore * 0.3 + 5; // ~5-35 range
  1351→  1351→    }
  1352→  1352→    const n = runningDeployments.length;
  1353→  1353→    return {
  1354→  1354→      cpu: cpu / n,
  1355→  1355→      memory: memory / n,
  1356→  1356→      network: network / n,
  1357→  1357→      disk: disk / n,
  1358→  1358→    };
  1359→  1359→  }, [runningDeployments]);
  1360→  1360→
  1361→  1361→  // Stable seed for initial series generation
  1362→  1362→  const seed = useMemo(
  1363→  1363→    () => `dash-${runningDeployments.map((d) => d.id).join(",")}`,
  1364→  1364→    [runningDeployments],
  1365→  1365→  );
  1366→  1366→
  1367→  1367→  // Tick counter drives series evolution
  1368→  1368→  const [tick, setTick] = useState(0);

  1372→  1372→    const t = setInterval(() => setTick((n) => n + 1), 10000);
  1373→  1373→    return () => clearInterval(t);
  1374→  1374→  }, []);

  1376→  1376→  // Compute full series from seed + tick (fully derived, no intermediate state)
  1377→  1377→  const series = useMemo(() => {
  1378→  1378→    const cpu = generateTimeSeries(aggMetrics.cpu, SPARKLINE_POINTS, 5, `${seed}:cpu`);
  1379→  1379→    const memory = generateTimeSeries(aggMetrics.memory, SPARKLINE_POINTS, 4, `${seed}:mem`);
  1380→  1380→    const network = generateTimeSeries(aggMetrics.network, SPARKLINE_POINTS, 6, `${seed}:net`);
  1381→  1381→    const disk = generateTimeSeries(aggMetrics.disk, SPARKLINE_POINTS, 4, `${seed}:disk`);
  1382→  1382→    // Apply tick-based evolution
  1383→  1383→    let c = cpu, m = memory, n = network, d = disk;
  1384→  1384→    for (let i = 0; i < tick; i++) {
  1385→  1385→      c = tickTimeSeries(c, aggMetrics.cpu, 5, i + 1);
  1386→  1386→      m = tickTimeSeries(m, aggMetrics.memory, 4, i + 1);
  1387→  1387→      n = tickTimeSeries(n, aggMetrics.network, 6, i + 1);
  1388→  1388→      d = tickTimeSeries(d, aggMetrics.disk, 4, i + 1);
  1389→  1389→    }
  1390→  1390→    return { cpu: c, memory: m, network: n, disk: d };
